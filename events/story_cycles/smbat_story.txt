# smbat_story.0101: Polemic against the Romans
# You're preaching against Roman imperial power
# If the holder of the current location is pro-roman this could piss them off
# But if they're anti-Roman it could actually help you
# Either way, preaching more aggressively raises Tondrakian Fervor more
smbat_story.0101 = {
	type = character_event
	title = smbat_story.0101.t
	desc = {
		desc = smbat_story.0101.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:local_character.top_participant_group:situation_caucasus = { participant_group_type = pro_roman } }
				desc = smbat_story.0101.desc.pro_roman
			}
			desc = smbat_story.0101.desc.anti_roman
		}
	}
	theme = faith
	left_portrait = {
		character = ROOT
		animation = debating 
	}
	right_portrait = {
		character = scope:local_character
		triggered_animation = {
			trigger = {
				NOT = { scope:local_character.top_participant_group:situation_caucasus = { participant_group_type = pro_roman } }
			}
			animation = throne_room_applaud_1
		}
		animation = disapproval
	}
	widget = {
		gui = "event_window_widget_smbat"
		container = "custom_widgets_container"
	}
	immediate = {
		smbat_of_zarehawan_widget_setup_effect = yes # setup widget
		# fetch characters
		if = {
			limit = { exists = current_travel_plan }
			current_travel_plan.current_location.county.holder = { save_scope_as = local_character }
		}
		else = {
			location.county.holder = { save_scope_as = local_character }
		}
	}

	option = {
		name = smbat_story.0101.a # preach aggressively
		change_spiritual_fulfillment = 5
		stress_and_fulfillment_impact = {
			brave = minor_stress_impact_loss
		}
		fulfillment_impact
		smbat_change_rebelliousness_effect = {
			VALUE = 5
		}
		reverse_add_opinion = {
			target = scope:local_character
			modifier = spoke_against_rome
			opinion = {
				value = 10
				if = {
					limit = { scope:local_character.top_participant_group:situation_caucasus = { participant_group_type = pro_roman } }
					multiply = -1
				}
			}
		}
	}

	option = {
		name = smbat_story.0102.b # preach moderately
		change_spiritual_fulfillment = -5
		stress = {
			craven = minor_stress_impact_loss
		}
		smbat_change_rebelliousness_effect = {
			VALUE = -5
		}
	}

	option = {
		name = smbat_story.0101.c # preach explosively
		change_spiritual_fulfillment = 10
		stress_and_fulfillment_impact = {
			brave = minor_stress_impact_loss
			zealous = minor_stress_impact_loss
		}
		smbat_change_rebelliousness_effect = {
			VALUE = 10
		}
		scope:the_faith = {
			change_fervor = {
				value = 5
			}
		}
		reverse_add_opinion = {
			target = scope:local_character
			modifier = spoke_against_rome
			opinion = {
				value = 20
				if = {
					limit = { scope:local_character.top_participant_group:situation_caucasus = { participant_group_type = pro_roman } }
					multiply = -1
				}
			}
		}
	}

}

# smbat_story.0102: debate with local clergy
# You are debating with a local clergyman about theology
# Depending on what you're going for and whether you win, you could:
# raise/lower Rebelliousness
# raise/lower Righteousness
# Possibly convert and recruit the clergyman to your court/camp
smbat_story.0102 = {
	type = character_event
	title = smbat_story.0102.t
	desc = smbat_story.0102.desc
	theme = faith
	left_portrait = {
		character = ROOT
		animation = debating
	}
	right_portrait = {
		character = scope:local_clergyman
		animation = debating
	}
	widget = {
		gui = "event_window_widget_smbat"
		container = "custom_widgets_container"
	}
	immediate = {
		smbat_of_zarehawan_widget_setup_effect = yes # setup widget
		# fetch character
		if = {
			limit = { exists = current_travel_plan }
			# If the holding is under lease to an order, get the ordermaster
			current_travel_plan.current_location = { save_scope_as = location }
		}
		else = {
			location = { save_scope_as = location }
		}
		# if the location is a temple holding...
		if = {
			limit = { scope:location = { has_holding_type = church_holding } }
			# Then we first check if it's leased and if so to whom?
			if = {
				limit = { exists = scope:location.barony.lessee }
				if = {
					limit = { scope:location.barony.lessee = { has_government = holy_order_government } }
					create_character = {
						template = holy_order_leader_character
						location = scope:location
						rite = scope:location.rite
						culture = scope:location.culture
						save_scope_as = local_clergyman
					}
				}
				else_if = {
					limit = { scope:location.barony.lessee = { government_has_flag = government_is_monastic_holy_order } }
					create_character = {
						template = monk_character_template
						location = scope:location
						rite = scope:location.rite
						culture = scope:location.culture
						save_scope_as = local_clergyman
					}
				}
			}
			# If it's neither a holy order not leased to the realm priest, we just gen a placeholder priest
			if = {
				limit = {NOT = { exists = scope:local_clergyman }}
				create_character = {
					template = priest_character_template
					location = scope:location
					rite = scope:location.rite
					culture = scope:location.culture
					save_scope_as = local_clergyman
				}
			}
		}

		# Debate the clergyman to try and convince him
		option = {
			name = smbat_story.0102.a
		}
		# Debate the clergyman to humiliate him
		option = {
			name = smbat_story.0102.b
		}
		# Debate the clergyman to try and find truth
		option = {
			name = smbat_story.0102.c
		}
	}
}
