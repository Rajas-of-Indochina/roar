on_death = {
	on_actions = {
		situation_caucasus_on_death
	}
}

situation_caucasus_on_death = {
	effect = {
		if = {
			limit = {
				# if ROOT was a landed ruler or adventurer participating in the South Caucasus Situation
				exists = scope:killer
				ROOT = {
					OR = {
						is_landed_or_landless_administrative = yes
						is_landless_adventurer = yes
					}
					any_character_situation = { situation_type = situation_caucasus }
				}
			}
			# Trigger the "character murdered" in subregion(s) based on de jure of primary title
			situation:situation_caucasus = {
				every_situation_sub_region = {
					limit = {
						any_situation_sub_region_county = {
							target_is_de_jure_liege_or_above = ROOT.primary_title
						}
					}
					trigger_sub_region_catalyst = participant_murdered
				}
			}
		}
	}
}

# Called when a raid action finishes
# root is the raid army
# scope:raider is the person owning the raid army
# scope:barony is the barony title that got raided
# scope:county is the county title for the barony
on_raid_action_completion = {
	on_actions = {
		situation_caucasus_raid_action_completed
	}
}

situation_caucasus_raid_action_completed = {
	trigger = {
		scope:county = { any_county_situation = { situation_type = situation_caucasus } } # Gotta be a county in the situation
	}
	effect = {
		situation:situation_caucasus = {
			trigger_caucasus_catalyst_in_title = {
				TITLE = scope:title
				CATALYST = holding_raided
			}
		}
	}
}

# called for when a casus belli resolves in one of the following ways.
# same scopes are available in these events as in the CBs themselves, this just fires for all CBs instead of a specific CB.
# note that any events/effects fired here WILL NOT show up in the war summary tooltip; they fire at the same time as the war resolution, but are not actually part of the war resolution itself.
on_war_won_attacker = {
	on_actions = {
		situation_caucasus_war_won_attacker
	}
}

situation_caucasus_war_won_attacker = {
	effect = {
		# Trigger disarray catalyst if involved character loses a border raid war
		if = {
			limit = {
				scope:war = { using_cb = fp2_border_raid }
				scope:defender = { any_character_situation = { situation_type = situation_caucasus } }
			}
			scope:war.casus_belli = {
				every_in_list = {
					list = target_titles
					limit = { tier = tier_county }
					save_scope_as = the_title
					situation:situation_caucasus = {
						trigger_caucasus_catalyst_in_title = {
							TITLE = scope:the_title
							CATALYST = raid_cb_enforced
						}
					}
				}
			}
		}
		# Victorious peasant rabble
		if = {
			limit = {
				scope:war = { using_cb = peasant_war }
				scope:defender = { any_character_situation = { situation_type = situation_caucasus } }
			}
			situation:situation_caucasus = {
				every_situation_sub_region = {
					limit = {
						any_situation_sub_region_county = {
							target_is_de_jure_liege_or_above = scope:defender.primary_title
						}
					}
					trigger_sub_region_catalyst = successful_peasant_revolt
				}
			}
		}
		# Victorious pro-Roman Populists
		if = {
			limit = {
				scope:war = { using_cb = populist_war }
				trigger_if = {
					limit = { scope:defender.faith = { rite_is_pro_roman_trigger = no } }
					scope:attacker.faith = { rite_is_pro_roman_trigger = yes }
				}
				trigger_else_if = {
					limit = { scope:defender.faith = { rite_is_pro_roman_trigger = yes } }
					scope:attacker.faith = { rite_is_pro_roman_trigger = yes }
					scope:attacker.culture = { culture_is_pro_roman_trigger = yes }
				}
			}
			situation:situation_caucasus = {
				every_situation_sub_region = {
					limit = {
						any_situation_sub_region_county = {
							target_is_de_jure_liege_or_above = scope:defender.primary_title
						}
					}
					trigger_sub_region_catalyst = successful_pro_roman_revolt
				}
			}
		}
		# Victorious pro-Perso-Arabic Populists
		if = {
			limit = {
				scope:war = { using_cb = populist_war }
				trigger_if = {
					limit = { scope:defender.faith = { rite_is_pro_perso_arabic_trigger = no } }
					scope:attacker.faith = { rite_is_pro_perso_arabic_trigger = yes }
				}
				trigger_else_if = {
					limit = { scope:defender.faith = { rite_is_pro_perso_arabic_trigger = yes } }
					scope:attacker.faith = { rite_is_pro_perso_arabic_trigger = yes }
					scope:attacker.culture = { culture_is_pro_perso_arabic_trigger = yes }
				}
			}
			situation:situation_caucasus = {
				every_situation_sub_region = {
					limit = {
						any_situation_sub_region_county = {
							target_is_de_jure_liege_or_above = scope:defender.primary_title
						}
					}
					trigger_sub_region_catalyst = successful_pro_perso_arabic_revolt
				}
			}
		}
		# Victorious Autonomist Populists
		if = {
			limit = {
				scope:war = { using_cb = populist_war }
				trigger_if = {
					limit = { scope:defender.faith = { rite_is_autonomist_trigger = no } }
					scope:attacker.faith = { rite_is_autonomist_trigger = yes }
				}
				trigger_else_if = {
					limit = { scope:defender.faith = { rite_is_autonomist_trigger = yes } }
					scope:attacker.faith = { rite_is_autonomist_trigger = yes }
					scope:attacker.culture = { culture_is_autonomist_trigger = yes }
				}
			}
			situation:situation_caucasus = {
				every_situation_sub_region = {
					limit = {
						any_situation_sub_region_county = {
							target_is_de_jure_liege_or_above = scope:defender.primary_title
						}
					}
					trigger_sub_region_catalyst = successful_autonomist_revolt
				}
			}
		}
	}
}

	on_actions = {
		situation_caucasus_on_war_won_defender
	}
}

situation_caucasus_on_war_won_defender = {
	effect = {
		if = {
			limit = {
				scope:defender.top_participant_group:situation_caucasus = { participant_group_type = interloper }
				scope:war = {
					OR = {
						using_cb = independence_faction_war
						using_cb = liberty_faction_war
						using_cb = claimant_faction_war
						using_cb = nation_fracturing_faction_war
						using_cb = nomadic_war
						using_cb = peasant_war
						using_cb = populist_war
						# using_cb = communalist_war - TODO
					}
				}
			}
			situation:situation_caucasus = {
				trigger_caucasus_catalyst_in_title = {
					TITLE = scope:defender.primary_title
					CATALYST = faction_against_interloper_failed
				}
			}
		}
		else_if = {
			limit = {
				scope:defender.top_participant_group:situation_caucasus = { participant_group_type = autonomist }
				NOT = { scope:attacker.top_participant_group:situation_caucasus = { participant_group_type = autonomist } }
			}
			situation:situation_caucasus = {
				trigger_caucasus_catalyst_in_title = {
					TITLE = scope:defender.primary_title
					CATALYST = autonomist_repelled_invasion
				}
			}
		}
	}
}

	}
}

# A title is transferred to a new character
# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Might be dead
on_title_gain = {
	on_actions = {
		situation_caucasus_on_title_gain_interloper
		situation_caucasus_on_title_gain_pro_roman
		situation_caucasus_on_title_gain_pro_perso_arabic
		situation_caucasus_on_title_gain_autonomist
	}
}

situation_caucasus_on_title_gain_interloper = {
	effect = {
		# Interloper seizes a title from a non-interloper (tiered)
		if = {
			limit = {
				scope:title.tier > tier_barony
				scope:title = {
					trigger_if = {
						limit = {tier = county }
						any_county_situation = { situation_type = situation_caucasus }
					}
					trigger_else = {
						any_de_jure_county = {
							any_county_situation = { situation_type = situation_caucasus }
						}
					}
				}
				scope:previous_holder = {
					any_character_situation = { situation_type = situation_caucasus }
					NOT = { top_participant_group:situation_caucasus ?= { participant_group_type = interloper } }
				}
				ROOT = {
					OR = {
						top_participant_group:situation_caucasus ?= { participant_group_type = interloper }
						NOT = { any_character_situation = { situation_type = situation_caucasus } }
					}
				}
				OR = {
					scope:transfer_type = flag:conquest
					scope:transfer_type = flag:conquest_holy_war
					scope:transfer_type = flag:conquest_claim
					scope:transfer_type = flag:conquest_populist
					scope:transfer_type = flag:usurped
					scope:transfer_type = flag:revoked
					scope:transfer_type = flag:faction_demand
				}
				switch = {
					trigger = scope:title.tier
					tier_county = {
						situation:situation_caucasus = {
							trigger_caucasus_catalyst_in_title = {
								TITLE = scope:title
								CATALYST = interloper_seized_title_county
							}
						}
					}
					tier_duchy = {
						situation:situation_caucasus = {
							trigger_caucasus_catalyst_in_title = {
								TITLE = scope:title
								CATALYST = interloper_seized_title_duchy
							}
						}
					}
					# treat anything bigger than a duchy as a kingdom
					fallback = {
						situation:situation_caucasus = {
							trigger_caucasus_catalyst_in_title = {
								TITLE = scope:title
								CATALYST = interloper_seized_title_kingdom
							}
						}
					}
				}
			}
		}
	}
}

# Code on-action: a new faith is created (character create/convert, or a rite detaching into its own faith)
# Root is the new faith; its head/leader is root.main_rite.head_of_rite (use it for faith-setup effects)
# scope:founder is the acting character that triggered the creation (creator/converter, heresy declarer, or on a divergence the head of the diverging rite)
# scope:founder can be absent - the game creates faiths with no acting character, so it is entered with ?=
# scope:old_faith is the faith they used to have
# scope:old_rite is the rite they used to have
# scope:diverged (true/false) optional scope, true if the faith was created because of a Rite diverging too much. Notice create a 100+ divergence heresy at the start does not count.
on_faith_created = {
	on_actions = {
		situation_caucasus_on_faith_created
	}
}

situation_caucasus_on_faith_created = {
	trigger = {
		OR = {
			# either scope:founder exists and is involved
			AND = {
				exists = scope:founder
				scope:founder ?= { any_character_situation = { situation_type = situation_caucasus } }
			}
			# or the faith is in the situation region
			AND = {
				NOT = { exists = scope:founder }
				any_faith_county = { title_province = { geographical_region = situation_caucasus_region } }
			}
		}
	}
	effect = {
		if = {
			limit = {
				OR = {
					scope:founder ?= { top_participant_group:situation_caucasus = { participant_group_type = pro_roman } } # Pro-Roman founder
					faith.main_rite = { rite_is_pro_roman_trigger = yes } # or the main rite is Pro-Roman
				}
			}
			situation:situation_caucasus ?= {
				trigger_caucasus_catalyst_in_faith = {
					FAITH = ROOT
					CATALYST = pro_roman_faith_created
				}
			}
		}
		if = {
			limit = {
				OR = {
					scope:founder ?= { top_participant_group:situation_caucasus = { participant_group_type = pro_perso_arabic } } # Pro-Roman founder
					faith.main_rite = { rite_is_pro_perso_arabic_trigger = yes } # or the main rite is Pro-Roman
				}
			}
			situation:situation_caucasus ?= {
				trigger_caucasus_catalyst_in_faith = {
					FAITH = ROOT
					CATALYST = pro_perso_arabic_faith_created
				}
			}
		}
		if = {
			limit = {
				OR = {
					scope:founder ?= { top_participant_group:situation_caucasus = { participant_group_type = autonomist } } # Pro-Roman founder
					faith.main_rite = { rite_is_autonomist_trigger = yes } # or the main rite is Pro-Roman
				}
			}
			situation:situation_caucasus ?= {
				trigger_caucasus_catalyst_in_faith = {
					FAITH = ROOT
					CATALYST = autonomist_faith_created
				}
			}
		}
	}
}
