setup_RICE_minorities = {
	trigger = {
		has_global_variable = is_RICE_loaded_global_variable
	}
	effect = {
		# Iceland
		every_county = {
			limit = {
				duchy = title:d_iceland
				game_start_date >= 850.1.1
				game_start_date <= 930.1.1
			}
			add_culture_minority_effect = {
				CULTURE = culture:irish
				SIZE = large
			}
			add_faith_minority_effect = {
				FAITH = faith:insular_celtic
				SIZE = large
			}
		}

		# Socotra
		title:c_socotra = {
			promote_faith_minority_effect = { FAITH = faith:shona_pagan }
			promote_culture_minority_effect = { CULTURE = culture:sindhi }
			promote_culture_minority_effect = { CULTURE = culture:shona }
			promote_culture_minority_effect = { CULTURE = culture:swahili }
		}

		# Yemen
		# Himyarite minority setup to be reworked

		# Horn of Africa
		title:c_burji = {
			add_culture_minority_effect = {
				CULTURE = culture:aari
				SIZE = small
			}
			add_faith_minority_effect = {
				FAITH = faith:south_omotic_pagan
				SIZE = small
			}
		}
		title:c_gamo = {
			add_culture_minority_effect = {
				CULTURE = culture:aari
				SIZE = small
			}
			add_faith_minority_effect = {
				FAITH = faith:south_omotic_pagan
				SIZE = small
			}
		}

		# Sicily
		# gonna just rip off India's religious diversity schtick for RICE Sicily's faiths and cultures
		every_county = {
			limit = { title_province = { geographical_region = custom_sicily } }
			every_neighboring_county = {
				if = {
					limit = { NOT = { faith = PREV.faith } }
					faith = { save_scope_as = faith }
					PREV = {
						add_faith_minority_effect = {
							FAITH = scope:faith
							SIZE = small
						}
					}
				}
				if = {
					limit = { NOT = { culture = PREV.culture } }
					culture = { save_scope_as = culture }
					PREV = {
						add_culture_minority_effect = {
							CULTURE = scope:culture
							SIZE = small
						}
					}
				}
			}
			if = {
				limit = { NOT = { faith = holder.faith } }
				add_faith_minority_effect = {
					FAITH = holder.faith
					SIZE = small
				}
			}
			if = {
				limit = { NOT = { culture = holder.culture } }
				add_culture_minority_effect = {
					CULTURE = holder.culture
					SIZE = small
				}
			}
		}
	}
}
