# A title is transferred to a new character
# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Might be dead
on_title_gain = {
	on_actions = {
		sea_stateless_title_handler
	}
}

on_game_start = {
	on_actions = {
		stateless_realm_cracker
	}
}

# Make sure that Stateless rulers can't gain titles.
# They already can't inherit them from dynastic govs
# this is to make sure that if they manage to seize any then they get handed out to confederate or destroyed
# if you see a stateless character with a title tier higher than count then something's gone hinky, amiss or perhaps even pear-shaped
sea_stateless_title_handler = {
	effect = {
		if = {
			limit = {
				government_has_flag = government_is_stateless
			}
			# No duchies, no kingdoms
			if = {
				limit = { scope:title.tier >= tier_duchy }
				destroy_title = scope:title
			}
			# Also no holding land outside your capital
			else_if = {
				limit = {
					NOT = { scope:title = ROOT.capital_county }
					NOT = {
						confederation = { has_cohesion_level_parameter = confederates_can_hold_multiple_counties }
					}
				}
				create_character = {
					location = scope:title.title_province
					template = generic_peasant_character
					faith = scope:title.faith
					culture = scope:title.culture
					save_scope_as = new_count
				}
				give_commune_county_effect = yes
			}
		}
	}
}

on_become_independent_after_grant_title_at_vassal_limit = {
	on_actions = { sea_mandala_tribal_wanua_fixer }
}

sea_mandala_tribal_wanua_fixer = {
	trigger = {
		scope:recipient = {
			government_has_flag = government_is_tribal
			capital_province = { geographical_region = world_asia_southeast_islands }
		}
	}
	effect = {
		scope:recipient = {
			change_government = wanua_government
		}
	}
}

# Happens when a county is granted to an auto-generated noble of the same culture as the ruler
# Root = The culture
# scope:actor = The granter
# scope:landed_title = The landed title
on_county_auto_granted_to_liege_culture = {
	on_actions = { sea_auto_grant_wanua_fixer }
}
on_county_auto_granted_to_local_culture = {
	on_actions = { sea_auto_grant_wanua_fixer }
}

sea_auto_grant_wanua_fixer = {
	trigger = {
		scope:landed_title.holder = {
			government_has_flag = government_is_tribal
			capital_province = { geographical_region = world_asia_southeast_islands }
		}
	}
	effect = {
		scope:landed_title.holder = {
			change_government = wanua_government
		}
	}
}

stateless_realm_cracker = {
	effect = {
		every_independent_ruler = {
			limit = {
				government_has_flag = government_is_stateless
				realm_size > 1
			}
			save_scope_as = ruler
			every_held_title = {
				limit = {
					tier >= tier_duchy
				}
				scope:ruler = { destroy_title = PREV }
			}
			capital_county = {
				duchy = {
					save_scope_as = capital_duchy
				}
			}
			create_confederation = {
				name = CONFEDERATION_DUCHY_NAME
				type = confederation_type:stateless_confederation
				leader = house
			}
			every_held_title = {
				limit = {
					tier = tier_county
					NOT = { scope:ruler.capital_county = this }
				}
				save_scope_as = title
				create_character = {
					location = scope:title.title_province
					template = generic_peasant_character
					faith = scope:title.faith
					culture = scope:title.culture
					save_scope_as = new_count
				}
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:title = {
					change_title_holder = {
						holder = scope:new_count
						change = scope:change
						take_baronies = yes
					}
				}
				resolve_title_and_vassal_change = scope:change
				scope:ruler = {
					create_or_add_to_stateless_confederation_effect = { TARGET = scope:new_count }
				}
			}
		}
	}
}
