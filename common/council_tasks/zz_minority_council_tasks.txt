	potential_county = {
		scope:county = {
			is_landless_type_title = no
			NOT = {

			}

			trigger_if = {
				limit = {
					scope:councillor_liege = {
						highest_held_title_tier >= tier_hegemony
						government_has_flag = government_has_merit
					}
				}
				faith != scope:councillor_liege.faith
			}
			#Otherwise it's councillor's faith
			trigger_else = { faith != scope:councillor.faith }
			trigger_if = {
				limit = {
					scope:councillor_liege = {
						is_alive = yes
						has_variable = hold_court_8120_religion_block
					}
				}
				custom_tooltip = {
					text = hold_court_8120_conversion_tt
					faith != scope:councillor_liege.var:hold_court_8120_religion_block
				}
			}
			custom_description = {
				text = "is_protected_via_contract_self_or_liege"
				subject = holder
				NAND = { # Vassal Contract forbids meddling by liege
					exists = holder.liege
					holder = {
						OR = {
							AND = {
								liege = scope:councillor_liege
								is_ruler = yes
								top_liege != this
								vassal_contract_has_flag = religiously_protected
							}
							any_liege_or_above = {
								liege ?= scope:councillor_liege
								is_ruler = yes
								top_liege != this
								vassal_contract_has_flag = religiously_protected
							}
						}
					}
				}
			}
			trigger_if = {
				limit = {	
					AND = {
						has_variable = cannot_be_converted_by_value
						var:cannot_be_converted_by_value = scope:councillor_liege		
					}
				}
				custom_tooltip = {
					text = promised_to_not_convert_county	
					NOR = {
						has_variable = cannot_be_converted_by_value
						var:cannot_be_converted_by_value = scope:councillor_liege

				}
			}
			# Unreformed Pagans do not tend to even try conversion of non-unreformed faiths unless zealous
			trigger_if = {
				limit = {
					scope:councillor_liege = {
						is_ai = yes
						faith = {
							has_doctrine_parameter = unreformed
						}
					}
				}
				OR = {
					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }

					scope:councillor_liege = { ai_zeal >= 50 }

						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith
							}
						}
					}
				}
			}
			# Do not convert Righteous faiths, unless disturbingly zealous
			trigger_if = {
				limit = {
					scope:councillor_liege = {
						is_ai = yes

							}
						}
					}
				}

					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
					}

						ai_zeal >= 100
						ai_rationality <= -50
					}
					scope:county = { # Always convert holy sites
						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith
							}
						}
					}
				}
			}
			# Converting Astray faiths only happens during certain circumstances

				limit = {
					scope:councillor_liege = {
						is_ai = yes
						faith = {
							faith_hostility_level = {
								target = scope:county.faith
								value <= faith_astray_level
							}
						}
					}
				}
				trigger_if = {
					limit = {
						scope:councillor_liege = {

						}
					}

						scope:county = { # Always convert holy sites
							any_county_province = {
								barony = {
									is_holy_site_of = scope:councillor_liege.faith
								}
							}
						}
					}
				}

					limit = {
						scope:councillor_liege = {
							faith = { has_doctrine = doctrine_pluralism_fundamentalist }
						}
					}
					OR = {
						scope:councillor_liege = {
							has_character_flag = vassal_directive_convert_faith
							vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
						}
						scope:councillor_liege = { ai_zeal >= 0 }
						scope:county = { county_opinion <= -10 }

								barony = {

						}
						scope:councillor_liege = { ai_zeal >= 25 }
						scope:county = { county_opinion <= -15 }
						scope:county = { # Always convert holy sites
							any_county_province = {
								barony = {
									is_holy_site_of = scope:councillor_liege.faith
								}
							}
						}
					}
				}
			}
			# Ash'ari in Spain tend to not convert Muwhalladi often

				limit = {
					scope:councillor_liege = {
						is_ai = yes
						faith = faith:ashari

						title_province = { geographical_region = world_europe_west_iberia }

				}
				OR = {
					scope:councillor_liege = {
						has_character_flag = vassal_directive_convert_faith
						vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }

					scope:county = { # Always convert holy sites
						any_county_province = {
							barony = {
								is_holy_site_of = scope:councillor_liege.faith

						}
					}

				limit = {
					scope:councillor_liege = {
						is_ai = yes

							OR = {
								has_doctrine = tenet_tax_nonbelievers

						}
					}

						AND = {
							scope:councillor_liege.religion = { is_in_family = rf_pagan }
							scope:county.religion = { is_in_family = rf_pagan }
						}
						AND = {
							scope:councillor_liege.religion = { is_in_family = rf_eastern }
							scope:county.religion = { is_in_family = rf_eastern }
						}
						AND = {
							scope:councillor_liege.religion = { is_in_family = rf_abrahamic }
							scope:county.religion = { is_in_family = rf_abrahamic }
						}
					}
				}

					scope:councillor_liege = {

						scope:county.title_province = {
							geographical_region = islamic_heartlands
						}

							}

							}

					}
				}

		}

				scope:councillor_liege.culture = {
					has_cultural_parameter = less_likely_to_promote_culture
				}
			}

		}

							is_hybrid_culture = yes
							is_divergent_culture = yes
						}
					}

							save_temporary_scope_as = culture_check
							scope:councillor_liege.culture = {
								any_parent_culture = {
									this = scope:culture_check
								}
							}
						}
					}
				}
			}
			add = 200
		}
		# Motivate the AI with an escalation agenda to take this task
		if = {
			limit = {
				has_character_flag = agenda_towards_escalation
			}
			add = 500
		}
		else_if = {
			limit = {
				has_character_flag = agenda_towards_deescalation
			}

							is_struggle_phase = struggle_iberia_phase_hostility
						}
					}
				}
			}
			multiply = 0
		}
		if = {
			# Interrupt this task if you're bankrupt
			limit = {
				scope:councillor_liege = {
					gold < -50
				}
			}
			multiply = 0
		}
		
		# AI directive
		if = {
			limit = {
				scope:councillor_liege = {
					has_character_flag = vassal_directive_convert_culture
					vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
				}
			}
			add = 50000
		}
	}

					}
				}
				# in which case we accept same heritage culture, and astray or better religions
				scope:county = {
					culture = { has_same_culture_heritage = scope:councillor_liege.culture }
					"faith_hostility_level(scope:councillor_liege.faith)" <= faith_astray_level
				}
			}
			add = 100
		}
		# or... if you're a German participating in Ostsiedlung
		if = {
			limit = {
				scope:councillor_liege = {
					culture = {
						has_cultural_tradition = tradition_east_settling
						has_cultural_pillar = heritage_central_germanic
					}
				}
				scope:county.culture = { has_cultural_pillar = heritage_central_germanic }
			}
			add = 50
		}
		# Or you're in maritime southeast asia and we want to simulate the Islamization of SEA
		if = {
			limit = {
				scope:councillor_liege.capital_county.title_province.geographical_region = world_asia_southeast_islands
				scope:county.faith.religion = religion:islam_religion
			}
			add = 50
		}
		# or you're a Crusader State in which case we REALLY want you to do this
		if = {
			limit = {
				OR = {
					scope:councillor_liege = { has_trait = crusader_king }
					scope:councillor_liege = { any_liege_or_above = { has_trait = crusader_king } }
				}
				scope:county.faith = scope:councillor_liege.faith
				scope:county.culture = scope:councillor_liege.culture
			}
			add = 200
		}
		# or you're bordering a county that's your culture or faith
		if = {
			limit = {
				OR = {
					scope:county.faith = scope:councillor_liege.faith
					scope:county.culture = scope:councillor_liege.culture
				}
				scope:county = {
					any_neighboring_county = { holder = scope:councillor_liege }
				}
			}
			add = 50
		}
		# or you're a Norman settling in Anglo-Saxon Britannia or Pre-melting pot Sicily
		if = {
			limit = {
				OR = {
					AND = {
						culture:english = { culture_number_of_counties = 0 }
						scope:councillor_liege.capital_county.title_province.geographical_region = world_europe_west_britannia
					}
					AND = {
						culture:sicilian = { culture_number_of_counties = 0 }
						scope:councillor_liege.capital_county.title_province.geographical_region = world_europe_south_italy
					}
				}
				scope:councillor.culture = culture:norman
				scope:county.culture = culture:norman
			}
			add = 50
		}
		# Or you're a in Hungary
		# in which case we want you to invite Germans
		# but not totally replace your own population
		# and only after they research Ostsiedlung
		if = {
			limit = {
				any_county_in_region = {
					holder = scope:councillor_liege
					NOT = {
						county_has_type_minority_with_trigger = {
							TYPE = culture
							SIZE = large
							TRIGGER = has_cultural_pillar
							CHECK = heritage_central_germanic
						}
						county_has_type_minority_with_trigger = {
							TYPE = culture
							SIZE = small
							TRIGGER = has_cultural_pillar
							CHECK = heritage_central_germanic
						}
					}
				}
				scope:county.culture = {
					has_cultural_pillar = heritage_central_germanic
					has_innovation = innovation_east_settling
				}
			}
			add = 50
		}
		# AI directive
		if = {
			limit = {
				OR = {
					AND = {
						scope:councillor_liege = {
							has_character_flag = vassal_directive_convert_culture
							vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
						}
						scope:county.culture = scope:councillor_liege.culture
					}
					AND = {
						scope:councillor_liege = {
							has_character_flag = vassal_directive_convert_faith
							vassal_is_valid_and_follows_directive_trigger = { VASSAL = this LIEGE = liege }
						}
						scope:county.faith = scope:councillor_liege.faith
					}
				}
			}
			add = 50000
		}
	}
}

			desc = task_invite_settlers_effect_desc
		}
		desc = {
			triggered_desc = {

					exists = scope:county
					scope:councillor_liege = {
						any_held_title = {
							title_tier = county
							NOT = { culture = scope:county.culture } }
						}
				}
				desc = task_invite_settlers_effect_culture_desc
			}
			triggered_desc = {
				trigger = {
					exists = scope:county
					scope:councillor_liege = {
						any_held_title = {
							title_tier = county
							NOT = { faith = scope:county.faith }
						}
					}
				}
				desc = task_invite_settlers_effect_faith_desc
			}
		}
	}

	progress = {
		value = 0
		add = {
			value = chancellor_invite_settlers_base
			desc = CHANCELLOR_INVITE_SETTLERS_PROGRESS_BASE
		}
		add = {
			value = scope:councillor.chancellor_invite_settlers_monthly_increase
			desc = SCALED_COUNCILLOR_DIPLOMACY_VALUE
		}

		add = chancellor_invite_settlers_contextual_bonuses

		# State Faith Boost
		if = {
			limit = {
				scope:councillor.top_liege.primary_title.state_faith ?= {
					this = scope:county.faith
					this = scope:councillor.top_liege.faith
				}
			}
			multiply = {
				value = ep3_state_faith_conversion_mult_value
				desc = ep3_convert_county_to_state_faith
			}
		}
		
		# Conversion speed game rules
		if = {
			limit = {
				OR = {
					has_game_rule = slower_faith_conversion_speed
					has_game_rule = slower_culture_conversion_speed
				}
			}
			multiply = {
				value = slower_game_rule_value
				desc = GAME_RULE_SLOWER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = significantly_slower_faith_conversion_speed
					has_game_rule = significantly_slower_culture_conversion_speed
				}
			}
			multiply = {
				value = significantly_slower_game_rule_value
				desc = GAME_RULE_SIGNIFICANTLY_SLOWER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = faster_faith_conversion_speed
					has_game_rule = faster_culture_conversion_speed
				}
			}
			multiply = {
				value = faster_game_rule_value
				desc = GAME_RULE_FASTER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = significantly_faster_faith_conversion_speed
					has_game_rule = significantly_faster_culture_conversion_speed
				}
			}
			multiply = {
				value = significantly_faster_game_rule_value
				desc = GAME_RULE_SIGNIFICANTLY_FASTER_REASON
			}
		}
		# Natural disaster resettlement
		if = {
			limit = {
				scope:councillor_liege = { any_character_situation = { situation_current_phase = recovery } }
			}
			multiply = {
				value = 2
				desc = DISASTER_RECOVERY_REASON
			}
		}
		# If you're inviting from a county that doesn't border on you hold then it's half speed
		if = {
			limit = {
				NOT = {
					scope:county = { any_neighboring_county = { holder = scope:councillor_liege } }
				}
			}
			multiply = {
				value = 0.5
				desc = NOT_BORDERING_REALM_REASON
			}
		}
        # Minorities: Three stages, triple speed, evens out to the same amount
        multiply = {
            value = 3
            desc = MINORITIES_SYSTEM_REASON
        }
		min = {
			value = 0.1
			desc = minimum_conversion
		}
	}

	potential_county = {
		scope:county = {
			is_landless_type_title = no
			NOT = { holder = scope:councillor_liege }
			holder = { in_diplomatic_range = scope:councillor_liege }
			# Kurultai version ONLY gets to bring in settlers that are their own culture/faith

			culture = scope:councillor_liege.faith

			custom_tooltip = {
				text = invite_settlers_different_faith_culture_tt
				any_held_title = {
					title_tier = county
					OR = {
						faith != scope:county.faith
						culture != scope:county.culture
					}
				}
			}
		}
	}

	on_start_task_county = {
		# this is to prevent everyone from inviting from fucking Paris
		set_variable = {
			name = inviter
			value = scope:councillor_liege
		}
	}

	on_finish_task_county = {
		scope:councillor_liege = {
			random_held_title = {
				# Prefer any county within a recovering natural disaster
				limit = {
					title_tier = county
					is_landless_type_title = no
					any_county_situation = { situation_current_phase = recovery }
					OR = {
						culture != scope:county.culture
						faith != scope:county.faith
					}
				}
				# failing that a county that is neither scope:county's faith or culture
				alternative_limit = {
					title_tier = county
					is_landless_type_title = no
					culture != scope:county.culture
					faith != scope:county.faith
				}
				# failing that a county that at least doesn't share one.
				alternative_limit = {
					title_tier = county
					is_landless_type_title = no
					OR = {
						faith != scope:county.faith
						culture != scope:county.culture
					}
				}
				promote_culture_minority_effect = { CULTURE = scope:county.culture }
				promote_faith_minority_effect = { FAITH = scope:county.faith }
				save_scope_value_as = {
					name = target_county_dev
					value = development_level
				}
				change_development_progress_with_overflow = {
					value = scope:county.development_level
					subtract = scope:target_county_dev
					multiply = 10
				}
			}
		}
		scope:county = {
			change_development_progress_with_overflow = {
				value = scope:target_county_dev
				subtract = development_level
				multiply = 10
			}
			remove_variable = inviter
		}
	}

	ai_will_do = {
		value = 0
		if = { limit = { scope:councillor = { is_performing_council_task = task_invite_settlers } } add = 1000 }
		# you do this if you're ruling over a lot of foreigners
		if = {
			limit = {
				scope:councillor_liege = {
					any_sub_realm_county = {
						percent >= 0.4
						faith != scope:councillor_liege.faith
						culture != scope:councillor_liege.culture
					}
				}
			}
			add = 100
		}
	}
}

			value = scope:councillor.chancellor_invite_settlers_monthly_increase
			desc = SCALED_COUNCILLOR_DIPLOMACY_VALUE
		}

		add = chancellor_invite_settlers_contextual_bonuses

		# State Faith Boost
		if = {
			limit = {
				scope:councillor.top_liege.primary_title.state_faith ?= {
					this = scope:county.faith
					this = scope:councillor.top_liege.faith
				}
			}
			multiply = {
				value = ep3_state_faith_conversion_mult_value
				desc = ep3_convert_county_to_state_faith
			}
		}
		
		# Conversion speed game rules
		if = {
			limit = {
				OR = {
					has_game_rule = slower_faith_conversion_speed
					has_game_rule = slower_culture_conversion_speed
				}
			}
			multiply = {
				value = slower_game_rule_value
				desc = GAME_RULE_SLOWER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = significantly_slower_faith_conversion_speed
					has_game_rule = significantly_slower_culture_conversion_speed
				}
			}
			multiply = {
				value = significantly_slower_game_rule_value
				desc = GAME_RULE_SIGNIFICANTLY_SLOWER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = faster_faith_conversion_speed
					has_game_rule = faster_culture_conversion_speed
				}
			}
			multiply = {
				value = faster_game_rule_value
				desc = GAME_RULE_FASTER_REASON
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = significantly_faster_faith_conversion_speed
					has_game_rule = significantly_faster_culture_conversion_speed
				}
			}
			multiply = {
				value = significantly_faster_game_rule_value
				desc = GAME_RULE_SIGNIFICANTLY_FASTER_REASON
			}
		}
		# Natural disaster resettlement
		if = {
			limit = {
				scope:councillor_liege = { any_character_situation = { situation_current_phase = recovery } }
			}
			multiply = {
				value = 2
				desc = DISASTER_RECOVERY_REASON
			}
		}
		# If you're inviting from a county that doesn't border on you hold then it's half speed
		if = {
			limit = {
				NOT = {
					scope:county = { any_neighboring_county = { holder = scope:councillor_liege } }
				}
			}
			multiply = {
				value = 0.5
				desc = NOT_BORDERING_REALM_REASON
			}
		}
        # Minorities: Three stages, triple speed, evens out to the same amount
        multiply = {
            value = 3
            desc = MINORITIES_SYSTEM_REASON
        }
		min = {
			value = 0.1
			desc = minimum_conversion
		}
	}

	potential_county = {
		scope:county = {
			is_landless_type_title = no
			NOT = { holder = scope:councillor_liege }
			holder = { in_diplomatic_range = scope:councillor_liege }
			# Kurultai version ONLY gets to bring in settlers that are their own culture/faith
			faith = scope:councillor_liege.faith
			culture = scope:councillor_liege.faith
		}
		scope:councillor_liege = {
			custom_tooltip = {
				text = invite_settlers_different_faith_culture_tt
				any_held_title = {
					title_tier = county
					OR = {
						faith != scope:county.faith
						culture != scope:county.culture
					}
				}
			}
		}
	}

	on_start_task_county = {
		# this is to prevent everyone from inviting from fucking Paris
		set_variable = {
			name = inviter
			value = scope:councillor_liege
		}
	}

	on_finish_task_county = {
		scope:councillor_liege = {
			random_held_title = {
				# Prefer any county within a recovering natural disaster
				limit = {
					title_tier = county
					is_landless_type_title = no
					any_county_situation = { situation_current_phase = recovery }
					OR = {
						culture != scope:county.culture
						faith != scope:county.faith
					}
				}
				# failing that a county that is neither scope:county's faith or culture
				alternative_limit = {
					title_tier = county
					is_landless_type_title = no
					culture != scope:county.culture
					faith != scope:county.faith
				}
				# failing that a county that at least doesn't share one.
				alternative_limit = {
					title_tier = county
					is_landless_type_title = no
					OR = {
						faith != scope:county.faith
						culture != scope:county.culture
					}
				}
				promote_culture_minority_effect = { CULTURE = scope:county.culture }
				promote_faith_minority_effect = { FAITH = scope:county.faith }
				save_scope_value_as = {
					name = target_county_dev
					value = development_level
				}
				change_development_progress_with_overflow = {
					value = scope:county.development_level
					subtract = scope:target_county_dev
					multiply = 10
				}
			}
		}
		scope:county = {
			change_development_progress_with_overflow = {
				value = scope:target_county_dev
				subtract = development_level
				multiply = 10
			}
			remove_variable = inviter
		}
	}

	ai_will_do = {
		value = 0
		if = { limit = { scope:councillor = { is_performing_council_task = task_invite_settlers } } add = 1000 }
		# you do this if you're ruling over a lot of foreigners
		if = {
			limit = {
				scope:councillor_liege = {
					any_sub_realm_county = {
						percent >= 0.4
						faith != scope:councillor_liege.faith
						culture != scope:councillor_liege.culture
					}
				}
			}
			add = 100
		}
	}
}
