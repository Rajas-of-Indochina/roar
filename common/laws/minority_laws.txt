religious_minorities_law = {
    default = religious_minorities_lenient
    flag = minorities_law
    flag = religious_minorities_law
    
    religious_minorities_harsh = {

		modifier = {
			different_faith_opinion = -10
			clergy_opinion = 10
			minority_opinion = -5
			zealot_opinion = 5
		}

        flag = reduced_faith_migrations
        flag = disable_minority_faith_promotion

        can_have = {
            can_have_minority_laws_trigger = yes
        }

        can_keep = {
            can_have_minority_laws_trigger = yes
        }

		pass_cost = {
			prestige = increase_crown_authority_prestige_cost
		}

		on_pass = {
			set_variable = {
				name = minority_law_cooldown
				years = @minority_law_cooldown_years
			}
		}

		ai_will_do = {
			value = -50

			# Factor for personality
			add = ai_zeal
			add = ai_vengefulness

			add = {
				value = ai_honor
				multiply = -1
			}
			add = {
				value = ai_compassion
				multiply = -1
			}
			if = {
				limit = {
					OR = {
						has_trait = holy_warrior
						has_trait = crusader_king
					}
				}
				add = 100
			}
			if = {
				limit = {
					faith = {
						has_doctrine = doctrine_pluralism_fundamentalist
					}
				}
				add = 25
			}
			if = {
				limit = {
					# Check to make sure that the HRE, ERE, theocrats and heads of faith are not going to be all uwu huggy wuggy with other faiths
					OR = {
						has_title = title:e_hre
						has_title = title:e_byzantium
						has_title = faith.religious_head_title
						has_government = theocracy_government
					}
				}
				add = 1000
			}
		}
        
    }
