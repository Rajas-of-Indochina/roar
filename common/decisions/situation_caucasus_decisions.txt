### SITUATION MECHANICS DECISIONS
situation_caucasus_choose_alignment_decision = {
	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 36
		kingdom = 36
		empire = 12
		hegemony = 12
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_foundation.dds"
	}
	# sort_order = -20
	decision_group_type = major

	cooldown = { years = 5 }

	is_shown = {
		any_character_situation = {
			situation_type = situation_caucasus
		}
		OR = {
			is_landed = yes
			is_landless_adventurer = yes
		}
	}

	widget = {
		gui = "decision_view_widget_generic_multichoice_with_effects"
		controller = decision_option_list_controller
		decision_to_second_step_button = "SELECT_ACTION"

		### Pro-Roman
		item = {
			value = pro_roman
			is_valid = {
				custom_tooltip = {
					text = already_in_this_movement_desc
					situation:situation_caucasus ?= {
						situation_participant_group:pro_roman = {
							NOT = { participant_group_has_character = root }
						}
					}
				}
			}
			current_description = choose_pro_roman_desc
			localization = choose_pro_roman_name
			icon = "gfx/interface/illustrations/struggle_backgrounds/armenian_struggle_phase_roman_dominance_bg.dds"
			ai_chance = {
				value = 0
				if = {
					limit = {
						# If you're of a Pro-Roman Rite account for Zeal
						rite = { rite_is_pro_roman_trigger = yes }
					}
					add = ai_zeal
				}
				if = {
					limit = {
						# More likely if you're of a Pro-Roman Culture
						culture = { culture_is_pro_roman_trigger = yes }
					}
					add = 25
				}
				if = {
					# If it looks like the Romans are ahead, greedy and dishonorable characters will switch
					limit = {
						situation:situation_caucasus = {
							calc_true_if = {
								amount >= 2
								situation_sub_region:armenia_proper = { sub_region_current_phase = situation_caucasus_roman_control }
								situation_sub_region:iberia = { sub_region_current_phase = situation_caucasus_roman_control }
								situation_sub_region:albania = { sub_region_current_phase = situation_caucasus_roman_control }
							}
						}
					}
					add = ai_greed
				}
				# Also account for how you feel about the Emperor
				if = {
					limit = {
						any_character = { is_pro_roman_emperor_trigger = yes }
					}
					random_character = {
						limit = { is_pro_roman_emperor_trigger = yes }
						save_temporary_scope_as = imperator
					}
					add = {
						value = "opinion(temp_scope:imperator)"
						multiply = 0.5
					}
				}
			}
		}

		### Pro-Perso-Arabic
		item = {
			value = pro_perso_arabic
			is_valid = {
				custom_tooltip = {
					text = already_in_this_movement_desc
					situation:situation_caucasus ?= {
						situation_participant_group:pro_perso_arabic = {
							NOT = { participant_group_has_character = root }
						}
					}
				}
			}
			current_description = choose_pro_perso_arabic_desc
			localization = choose_pro_perso_arabic_name
			icon = "gfx/interface/illustrations/struggle_backgrounds/armenian_struggle_phase_persian_dominance_bg.dds"
			ai_chance = {
				value = 0
				if = {
					limit = {
						# If you're of a Pro-Perso-Arabic Rite account for Zeal
						rite = { rite_is_pro_perso_arabic_trigger = yes }
					}
					add = ai_zeal
				}
				if = {
					limit = {
						# More likely if you're of a Pro-Perso-Arabic Culture
						culture = { culture_is_pro_perso_arabic_trigger = yes }
					}
					add = 25
				}
				if = {
					# If it looks like the Persians and Arabis are ahead, greedy characters will switch
					limit = {
						situation:situation_caucasus = {
							calc_true_if = {
								amount >= 2
								situation_sub_region:armenia_proper = { sub_region_current_phase = situation_caucasus_perso_arabic_control }
								situation_sub_region:iberia = { sub_region_current_phase = situation_caucasus_perso_arabic_control }
								situation_sub_region:albania = { sub_region_current_phase = situation_caucasus_perso_arabic_control }
							}
						}
					}
					add = ai_greed
				}
				# Also account for how you feel about the Caliph or Shah
				if = {
					limit = {
						any_character = {
							OR = {
								has_title = title:e_arabia
								has_title = title:h_dar_al_islam
							}
						}
					}
					random_character = {
						limit = {
							OR = {
								has_title = title:e_arabia
								has_title = title:h_dar_al_islam
							}
						}
						save_temporary_scope_as = caliph
					}
					add = {
						value = "opinion(temp_scope:caliph)"
						multiply = 0.5
					}
				}
				if = {
					limit = {
						title:e_persia.holder = exists
						NOR = {
							title:e_persia.holder = title:e_arabia.holder
							title:e_persia.holder = title:h_dar_al_islam.holder
						}
					}
					add = {
						value = "opinion(title:e_persia.holder)"
						multiply = 0.5
					}
				}
			}
		}

		### Autonomist
		item = {
			value = autonomist
			is_valid = {
				custom_tooltip = {
					text = already_in_this_movement_desc
					situation:situation_caucasus ?= {
						situation_participant_group:autonomist = {
							NOT = { participant_group_has_character = root }
						}
					}
				}
			}
			current_description = choose_autonomist_desc
			localization = choose_autonomist_name
			flat = yes
			ai_chance = {
				value = 0
				if = {
					limit = {
						# If you're of a Autonomist Rite account for Zeal
						rite = { rite_is_autonomist_trigger = yes }
					}
					add = ai_zeal
				}
				if = {
					limit = {
						# More likely if you're of a Autonomist Culture
						culture = { culture_is_autonomist_trigger = yes }
					}
					add = 25
				}
				# Bold characters are more likely to go this way
				add = ai_boldness
			}
		}
	}

	cost = {
		influence = {
			value = 0
			if = {
				limit = {
					government_allows = administrative
				}
				# Base cost
				value = monumental_influence_value
				# Less expensive to pick a side as an interloper
				if = {
					limit = {
						situation:situation_caucasus ?= {
							situation_participant_group:interloper = {
								participant_group_has_character = root
							}
						}
					}
					multiply = 0.5
				}
			}
		}
		# Non-admins pay prestige instead
		prestige = {
			value = 0
			if = {
				limit = {
					NOT = { government_allows = administrative }
				}
				# Base cost
				value = {
					value = monumental_influence_value
					multiply = 2
				}
				# Less expensive to pick a side as an interloper
				if = {
					limit = {
						situation:situation_caucasus ?= {
							situation_participant_group:interloper = {
								participant_group_has_character = root
							}
						}
					}
					multiply = 0.5
				}
			}
		}
	}

	effect = {
		switch = {
			trigger = yes
			scope:pro_roman = {
				set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = yes }
				if = {
					limit = {
						is_in_confederation = yes
					}
					confederation = { remove_confederation_member = ROOT }
				}
			}
			scope:pro_perso_arabic = {
				set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = yes }
				if = {
					limit = {
						is_in_confederation = yes
					}
					confederation = { remove_confederation_member = ROOT }
				}
			}
			scope:autonomist = {
				set_caucasus_alignment = { FLAG = autonomist COOLDOWN = yes }
			}
		}
	}
}

situation_caucasus_align_culture_decision = {
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 30
		empire = 12
		hegemony = 12
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_foundation.dds"
	}
	decision_group_type = major

	is_shown = {
		any_character_situation = {
			situation_type = situation_caucasus
		}
		NOT = {
			check_caucasus_alignment_trigger = { ALIGN = interloper }
		}
		is_landed = yes
	}

	is_valid = {
		# use a global cooldown to prevent tug-of-wars
		custom_tooltip = {
			text = sway_culture_global_cooldown_tt
			NOT = {
				culture = { has_variable = sway_culture_cooldown }
			}
		}
		custom_tooltip = {
			text = you_are_the_culture_head
			culture = {
				culture_head = ROOT
			}
		}

		trigger_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = pro_roman } }
			culture = {
				custom_tooltip = {
					text = not_culture_is_pro_roman_trigger_tt
					subject = THIS
					culture_is_pro_roman_trigger = no
				}
			}
		}
		trigger_else_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = pro_perso_arabic } }
			culture = {
				custom_tooltip = {
					text = not_culture_is_pro_perso_arabic_trigger_tt
					subject = THIS
					culture_is_pro_perso_arabic_trigger = no
				}
			}
		}
		trigger_else_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = autonomist } }
			culture = {
				custom_tooltip = {
					text = not_culture_is_autonomist_trigger_tt
					subject = THIS
					culture_is_autonomist_trigger = no
				}
			}
		}
	}

	cost = {
		piety = massive_piety_value
	}

	effect = {
		culture = {
			if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = pro_roman } } }
				align_culture_effect = {
					TARGET = THIS
					FLAG = pro_roman
				}
				situation:situation_caucasus ?= {
					trigger_caucasus_catalyst_in_culture = {
						CULTURE = ROOT.culture
						CATALYST = culture_swayed_pro_roman
					}
				}
			}
			else_if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = pro_perso_arabic } } }
				align_culture_effect = {
					TARGET = THIS
					FLAG = pro_perso_arabic
				}
				situation:situation_caucasus ?= {
					trigger_caucasus_catalyst_in_culture = {
						CULTURE = ROOT.culture
						CATALYST = culture_swayed_pro_perso_arabic
					}
				}
			}
			else_if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = autonomist } } }
				align_culture_effect = {
					TARGET = THIS
					FLAG = autonomist
				}
				situation:situation_caucasus ?= {
					trigger_caucasus_catalyst_in_culture = {
						CULTURE = ROOT.culture
						CATALYST = culture_swayed_autonomist
					}
				}
			}
		}
	}
}

		kingdom = 30
		empire = 12
		hegemony = 12
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp3/fp3_decision_foundation.dds"
	}
	decision_group_type = major

	is_shown = {
		any_character_situation = {
			situation_type = situation_caucasus
		}
		NOT = {
			check_caucasus_alignment_trigger = { ALIGN = interloper }
		}
		is_landed = yes
	}

	is_valid = {
		# use a global cooldown to prevent tug-of-wars
		custom_tooltip = {
			text = sway_rite_global_cooldown_tt
			NOT = {
				rite = { has_variable = sway_rite_cooldown }
			}
		}
		# If you are head of rite, then you can just declare it
		# If you are not head of rite, then you must have either a strong hook on your HoR or have puppeted them
		trigger_if = {
			limit = { NOT = { rite.head_of_rite = ROOT } }
			if = {
				limit = {
					NOR = {
						has_strong_usable_hook = rite.head_of_rite
						NOT = { any_character_situation = { situation_type = situation_caucasus } }
					}
				}
			}
			rite.head_of_rite = {
				top_participant_group:situation_caucasus = ROOT.top_participant_group:situation_caucasus
			}
			OR = {
				rite.head_of_rite = {
					AND = {
						puppeteer ?= ROOT
						OR = {
							is_puppet_of_type = theological_agent_puppet
							is_puppet_of_type = external_theological_agent_puppet
						}
					}
				}
				has_strong_usable_hook = rite.head_of_rite
			}
		}

		trigger_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = pro_roman } }
			faith = {
				custom_tooltip = {
					text = not_rite_is_pro_roman_trigger_tt
					subject = THIS
					rite_is_pro_roman_trigger = no
				}
			}
		}
		trigger_else_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = pro_perso_arabic } }
			faith = {
				custom_tooltip = {
					text = not_rite_is_pro_perso_arabic_trigger_tt
					subject = THIS
					rite_is_pro_perso_arabic_trigger = no
				}
			}
		}
		trigger_else_if = {
			limit = { check_caucasus_alignment_trigger = { ALIGN = autonomist } }
			faith = {
				custom_tooltip = {
					text = not_rite_is_autonomist_trigger_tt
					subject = THIS
					rite_is_autonomist_trigger = no
				}
			}
		}
	}

	cost = {
		piety = massive_piety_value
	}

	effect = {
		if = {
			limit = {
				exists = rite.head_of_rite
				NOT = {
					rite.head_of_rite = ROOT
					rite.head_of_rite = { puppeteer = ROOT }
				}
				has_strong_usable_hook = rite.head_of_rite
			}
			use_strong_hook = rite.head_of_rite
		}
		faith = {
			if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = pro_roman } } }
				align_rite_effect = {
					TARGET = THIS
					FLAG = pro_roman
				}
				situation:situation_caucasus ?= {
					trigger_caucasus_catalyst_in_rite = {
						RITE = ROOT.rite
						CATALYST = rite_swayed_pro_roman
					}
				}
			}
			else_if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = pro_perso_arabic } } }
				align_rite_effect = {
					TARGET = THIS
					FLAG = pro_perso_arabic
				}
				situation:situation_caucasus ?= {
					trigger_caucasus_catalyst_in_rite = {
						RITE = ROOT.rite
						CATALYST = rite_swayed_pro_perso_arabic
					}
				}
			}
			else_if = {
				limit = { ROOT = { check_caucasus_alignment_trigger = { ALIGN = autonomist } } }

					TARGET = THIS
					FLAG = autonomist
				}

					trigger_caucasus_catalyst_in_rite = {
						RITE = ROOT.rite
						CATALYST = rite_swayed_autonomist
					}
				}
			}
		}
	}

	}

			capital_county.title_province = {
				geographical_region = custom_armenia_proper

				has_cultural_pillar = heritage_caucasian

			}
		}
		title:k_armenian_principality = {
			is_titular = yes
		}
	}
	is_valid = {
		is_independent_ruler = yes
		OR = {
			culture = {
				this = culture:armenian
			}
			culture = {
				any_parent_culture_or_above = {
					this = culture:armenian
				}
			}
		}
		OR = {
			has_title = title:k_vaspurakan
			completely_controls = title:k_vaspurakan
		}
		OR = {
			has_title = title:k_ayrarat
			completely_controls = title:k_ayrarat
		}
		trigger_if = {
			limit = {
				exists = title:k_armenian_principality.holder
			}
			has_title = title:k_armenian_principality
		}
		OR = {
			highest_held_title_tier <= tier_duchy
			has_primary_title = title:k_vaspurakan
			has_primary_title = title:k_armenian_principality
			has_primary_title = title:k_ayrarat
			has_primary_title = title:k_syunik
		}
		trigger_if = {
			limit = {
				has_legitimacy = yes
			}
			legitimacy_level >= 4
		}
		trigger_else_if = {
			limit = { government_has_flag = government_has_influence }
			influence_level >= 4
		}
		trigger_else = {
			prestige_level >= 4
		}
	}
	effect = {
		show_as_tooltip = {
			unify_armenian_principalities_decision_effect = yes
		}
		hidden_effect = {
			save_scope_as = founder
			trigger_event = {
				id = caucasianroa.016
			}
		}
	}
	ai_will_do = {
		base = 100
	}
}
