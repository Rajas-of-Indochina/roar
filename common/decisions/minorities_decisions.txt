invite_islamic_settlers_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	ai_check_interval = 48
	decision_group_type = minorities
	desc = invite_islamic_settlers_decision_desc

	is_shown = {
		exists = capital_county
		capital_county.title_province = {
			geographical_region = world_asia_southeast_islands
		}
	}

	is_valid = {
		custom_tooltip = {
			text = available_islamic_settlers_tt
			# need to have counties within the proper range
			# that are either majority-Muslim or have a large Muslim minority
			any_county = {
				save_temporary_scope_as = county
				"scope:county.title_province.squared_distance(root.capital_province)" <= squared_distance_monstrous
				any_county_province = {
					is_coastal = yes
					NOT = { geographical_region = world_asia_southeast_islands }
				}
				OR = {
					religion = religion_islam_religion
					county_has_type_minority_with_trigger = {
						TYPE = faith
						SIZE = large
						TRIGGER = religion
						CHECK = religion:islam_religion
					}
				}
			}
		}
	}

	minimum_cost = {
		piety = massive_piety_gain
		gold = {
			value = 0
			if = {
				limit = { culture = { has_cultural_era_or_later = culture_era_late_medieval } }
				add = massive_gold_value
			}
			else = {
				add = major_gold_value
			}
		}
	}

	is_valid = {
		OR = {
			religion = religion:islam_religion
			NOT = { has_trait = zealous }
			faith = {
				has_doctrine = tenet_islamic_syncretism
			}
		}
		NOT = {
			has_realm_law_flag = increased_faith_migrations
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	effect = {
		custom_tooltip = invite_islamic_settlers_decision_tt_a
		if = {
			limit = {
				is_ai = no
			}
			trigger_event = minorities.0401
		}
		else = {
			# simplified version
			every_held_title = {
				title_tier = county
				save_scope_as = target
				random_county = {
					limit = {
						save_temporary_scope_as = county
						"scope:county.title_province.squared_distance(root.capital_province)" <= squared_distance_monstrous # This will let the western Maritime SEA reach the Horn of Africa
						is_coastal_county = yes
						title_province = {
							NOT = { geographical_region = world_asia_southeast_islands }
						}
						OR = {
							religion = religion:islam_religion
							county_has_type_minority_with_trigger = {
								TYPE = faith
								SIZE = large
								TRIGGER = religion
								CHECK = religion:islam_religion
							}
						}
					}
					save_scope_as = source
					if = {
						limit = { religion = religion:islam_religion }
						faith = { save_scope_as = islamic_faith }
					}
					else = {
						random_in_list = {
							variable = faith_minorities_large
							limit = { religion = religion:islam_religion }
							save_scope_as = islamic_faith
						}
					}
					# just in case i fucked up
					if = {
						NOT = { exists = scope:islamic_faith }
						faith:ashari = { save_scope_as = islamic_faith }
					}
				}
				faith_migration_effect = {
					FAITH = scope:islamic_faith
					TARGET_COUNTY = scope:target
					SOURCE_COUNTY = scope:source
				}
				if = {
					limit = {
						county_has_specific_small_faith_minority = { FAITH = $FAITH$ }
					}
					add_county_modifier = {
						modifier = muslim_settlers_modifier_small
						years = 5
					}
				}
				else_if = {
					limit = {
						county_has_specific_large_faith_minority = { FAITH = $FAITH$ }
					}
					add_county_modifier = {
						modifier = muslim_settlers_modifier_large
						years = 5
					}
				}
				if = {
					limit = {
						faith = $FAITH$
					}
					add_county_modifier = {
						modifier = muslim_settlers_modifier_majority
						years = 5
					}
				}
			}
		}
	}

	cooldown = {
		years = 10
	}
}
