centralize_stateless_confederation_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	ai_check_interval = 60

	is_shown = {
		is_independent_ruler = yes
		government_has_flag = government_is_stateless
		is_confederation_member = yes
		confederation ?= { confederation_type = confederation_type:stateless_confederation }
	}

	is_valid = {
		influence_level >= 4
		confederation ?= {
			has_cohesion_level_parameter = confederation_can_tribalize
			any_confederation_member = {
				count = all
				OR = {
					custom_description = {
						text = "You have an available [strong_hook|E]"
						ROOT = { has_strong_usable_hook = PREV }
					}
					opinion = {
						target = ROOT
						value >= very_high_positive_opinion
					}
					NOT = { capital_county.duchy = ROOT.capital_county.duchy }
					hidden_trigger = {
						THIS = ROOT
					}
				}
			}
		}
		NOT = { is_title_created = capital_county.duchy }
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_at_war = no
	}

	effect = {
		centralize_stateless_confederation_effect = yes
	}

	ai_potential = {
		always = no
	}
}
