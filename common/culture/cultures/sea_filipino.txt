buranun = {
	# represents Buranun, Tagimaha and Baklaya - indigenous people of Sulu archipelago
	color = buranun

	ethos = ethos_bellicose
	heritage = heritage_filipino
	language = language_filipino
	martial_custom = martial_custom_equal
	head_determination = head_determination_domain

	traditions = {
		tradition_SEA_jungle_warfare 	# msl_note: generic Filipino tradition
		tradition_fishermen 			# msl_note: to represent the coastal populations
		tradition_culture_blending 		# msl_note: to represent their future assimilation into Tausug population
		tradition_religion_blending 	# msl_note: to represent their future conversion to Islam under Sulu Sultanate
		tradition_quarrelsome 			# msl_note: to represent their conflicts with Orang Dampuan Chams and other later arrivals to the archipelago
	}

	# msl_note: tradition_seafaring/tradition_maritime_way_of_life was removed for Buranun as it was more so Tausug who established Sulu Sultanate

	name_list = name_list_filipino

	coa_gfx = { sulu_coa_gfx filipino_coa_gfx austronesian_group_coa_gfx burman_group_coa_gfx }
	building_gfx = { southeast_asian_building_gfx }
	clothing_gfx = { mindanaoan_clothing_gfx malay_clothing_gfx southeast_asian_clothing_gfx }
	unit_gfx = { southeast_asian_unit_gfx }

	ethnicities = {
		100 = asian_austronesian
	}
}
