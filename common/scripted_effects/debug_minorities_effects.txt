debug_clear_minorities = {
    clear_variable_list = rite_minorities_small
    clear_variable_list = rite_minorities_large
    clear_variable_list = culture_minorities_small
    clear_variable_list = culture_minorities_large
}

debug_spawn_rite_minority = {
    rite = { save_scope_as = old_rite }
    random_rite_type = {
        limit = { NOT = { this = scope:old_rite } }
        save_scope_as = rite
    }
    promote_rite_minority_effect = {
        RITE = scope:rite
    }
}

debug_spawn_culture_minority = {
    random_culture_global = {
        limit = { NOT = { this = PREV.culture } }
        save_scope_as = culture
    }
    custom_tooltip = "[culture.GetName]"
    # promote_culture_minority_effect = {
    #     CULTURE = scope:culture
    # }
    promote_culture_minority_effect = { CULTURE = scope:culture }
}

# Run in County Scope
debug_rite_migration_test = {
    random_in_list = {
        variable = rite_minorities_small
        save_scope_as = migrating_rite
    }
    custom_tooltip = "[migrating_rite.GetName]"
    random_neighboring_county = {
        limit = { NOT = { rite = scope:migrating_rite } }
        save_scope_as = target_county
    }
    custom_tooltip = "[target_county.GetName]"
    rite_migration_effect = {
        RITE = scope:migrating_rite
        TARGET_COUNTY = scope:target_county
        SOURCE_COUNTY = THIS
    }
}
