# These are the core scripted effects of the system, with a version of each for faiths and cultures.

# add_x_minority_effect accepts two parameters:
# SIZE (large or small)
# CULTURE/FAITH (the scope you want to add as a minority)
# DO NOT USE THE CULTURE COMMANDS FOR FAITHS AND VICE VERSA, THEY WILL FUCK UP

# delete_x_minority_effect:
# takes the same params as its add_ counterpart. Deletes the minority from the list if present.

# promote_x_minority_effect:
# takes one parameter:
# FAITH/CULTURE: the scope being promoted.
# Automatically detects if a minority of that type is already present and then upgrades it if present
# No minority -> small minority -> large minority -> majority
# the on_actions should automatically handle logic for creating minorities on a county conversion


add_faith_minority_effect = {
    if = {
        limit = {
            NOR = {
                faith = $FAITH$
                county_has_specific_faith_minority = { FAITH = $FAITH$ }
            }
        }
        add_to_variable_list = {
            name = faith_minorities_$SIZE$
            target = $FAITH$
        }
    }
}

delete_faith_minority_effect = {
    remove_list_variable = {
        name = faith_minorities_$SIZE$
        target = $FAITH$
    }
}

promote_faith_minority_effect = {
    faith = { save_scope_as = old_faith }
    if = {
        limit = { faith = $FAITH$ }
        debug_log = "whuh-oh! FAITH Promotion had a fucky-wucky!"
    }
    else_if = {
        limit = { county_has_specific_small_faith_minority = { FAITH = $FAITH$ } }
        delete_faith_minority_effect = { FAITH = $FAITH$ SIZE = small }
        add_faith_minority_effect = { FAITH = $FAITH$ SIZE = large }
    }
    else_if = {
        limit = { county_has_specific_large_faith_minority = { FAITH = $FAITH$ } }
        delete_faith_minority_effect = { FAITH = $FAITH$ SIZE = large }
        set_county_faith = $FAITH$
    }
    else = {
        add_faith_minority_effect = { FAITH = $FAITH$ SIZE = small }
    }
}

demote_faith_minority_effect = {
    if = {
        limit = { county_has_specific_small_faith_minority = { FAITH = $FAITH$ } }
        delete_faith_minority_effect = { FAITH = $FAITH$ SIZE = small }
    }
    else_if = {
        limit = { county_has_specific_large_faith_minority = { FAITH = $FAITH$ } }
        delete_faith_minority_effect = { FAITH = $FAITH$ SIZE = large }
        add_faith_minority_effect = { FAITH = $FAITH$ SIZE = small }
    }
}

# run in county scope
# saves scope:culture which is a random culture picked from either the majority or a minority culture
get_random_county_culture_effect = {
    random_list = {
        4 = {
            trigger = { always = $MAJORITY$ }
            culture = { save_scope_as = culture }
        }
        1 = {
            trigger = { county_has_small_culture_minority = yes }
            modifier = {
                factor = "list_size(culture_minorities_small)"
            }
            random_in_list = {
                variable = culture_minorities_small
                save_scope_as = culture
            }
        }
        2 = {
            trigger = { county_has_large_culture_minority = yes }
            modifier = {
                factor = "list_size(culture_minorities_large)"
            }
            random_in_list = {
                variable = culture_minorities_large
                save_scope_as = culture
            }
        }
    }
}

# CULTURE = the culture to migrate
# TARGET_COUNTY = county to migrate to
# SOURCE_COUNTY = county to migrate from
culture_migration_effect = {
    $CULTURE$ = { save_scope_as = loc_culture }
    $TARGET_COUNTY$ = { save_scope_as = loc_target }
    $SOURCE_COUNTY$ = { save_scope_as = loc_source }
    if = {
        limit = { $TARGET_COUNTY$.culture = $CULTURE$ }
        debug_log = "ERROR: Trying to migrate minority culture into same-culture target"
    }
    $TARGET_COUNTY$ = {
        promote_culture_minority_effect = { CULTURE = $CULTURE$ }
        holder = {
            if = {
                limit = { NOT = { THIS = $SOURCE_COUNTY$.holder } }
                send_interface_message = {
                    type = msg_culture_conversion
                    title = msg_culture_migration
                    desc = msg_culture_migration_desc
                    left_icon = $SOURCE_COUNTY$
                    right_icon = $TARGET_COUNTY$
                }
            }
        }
    }
    $SOURCE_COUNTY$ = {
        demote_culture_minority_effect = { CULTURE = $CULTURE$ }
        holder = {
            send_interface_message = {
                type = msg_culture_conversion
                title = msg_culture_migration
                desc = msg_culture_migration_desc
                left_icon = $SOURCE_COUNTY$
                right_icon = $TARGET_COUNTY$
            }
        }
    }
}

# called from county scope
# OLD_MINORITY = the minority culture to check for
# NEW_MINORITY = the minority to replace it with
# WARNING: THIS WILL PROMOTE, NOT DIRECT ADD.
replace_culture_minority_effect = {
    if = {
        limit = {
            county_has_specific_small_culture_minority = { CULTURE = $OLD_MINORITY$ }
        }
        delete_culture_minority_effect = { SIZE = small CULTURE = $OLD_MINORITY$ }
        promote_culture_minority_effect = { CULTURE = $NEW_MINORITY$ }
    }
    else_if = {
        limit = {
            county_has_specific_large_culture_minority = { CULTURE = $OLD_MINORITY$ }
        }
        delete_culture_minority_effect = { SIZE = small CULTURE = $OLD_MINORITY$ }
        promote_culture_minority_effect = { CULTURE = $NEW_MINORITY$ }
        promote_culture_minority_effect = { CULTURE = $NEW_MINORITY$ }
    }
}

# COUNTY = county doing the expelling
# FAITH = faith being expelled
# CHARACTER = character doing the expulsion
expel_county_faith_minority_effect = {
    $COUNTY$ = {
        if = {
            limit = { county_has_specific_faith_minority = { FAITH = $FAITH$ } }
            $CHARACTER$ = {
                add_gold = $COUNTY$.development_level
                add_tyranny = {
                    value = $COUNTY$.development_level
                    divide = 10
                    min = 1
                }
            }
            change_development_progress_with_overflow = -50
            find_faith_migration_target_effect = { FAITH = $FAITH$ }
            faith_migration_effect = {
                SOURCE_COUNTY = $COUNTY$
                TARGET_COUNTY = scope:target_county
                FAITH = $FAITH$
            }
        }
    }
}

assimilate_cultures_into_child_culture_effect = {
    culture = { save_scope_as = child_culture }
    every_in_list = {
        variable = culture_minorities_small
        limit = {
            save_temporary_scope_as = parent_culture
            scope:child_culture = { any_parent_culture = { THIS = scope:parent_culture } }
        }
        delete_culture_minority_effect = {
            SIZE = small
            CULTURE = THIS
        }
    }
    every_in_list = {
        variable = culture_minorities_large
        limit = {
            save_temporary_scope_as = parent_culture
            scope:child_culture = { any_parent_culture = { THIS = scope:parent_culture } }
        }
        delete_culture_minority_effect = {
            SIZE = small
            CULTURE = THIS
        }
    }
}

# Region-to-region migration for when you want a more refined, targeted form of migration.
# Making this now so it'll be easy to use for RICE stuff later
# SOURCE_REGION
# TARGET_REGION

quick_region_migration_effect = {
    random_county_in_region = {
        region = $SOURCE_REGION$
        save_scope_as = source_county
        get_random_county_culture_effect = { MAJORITY = yes } # scope:culture now saved
        get_random_county_faith_effect = { MAJORITY = yes } # scope:faith now saved
    }
    random_county_in_region = {
        region = $TARGET_REGION$
        limit = { NOT = { this = scope:source_county } } # with this we can support using this intra-regionally
        save_scope_as = target_county
    }
    combined_migration_effect = {
        SOURCE_COUNTY = scope:source_county
        TARGET_COUNTY = scope:target_county
        FAITH = scope:faith
        CULTURE = scope:culture
    }
}

targeted_regional_migration_effect = {
    random_county_in_region = {
        region = $SOURCE_REGION$
        save_scope_as = source_county
        get_random_county_culture_effect = { MAJORITY = yes } # scope:culture now saved
        get_random_county_faith_effect = { MAJORITY = yes } # scope:faith now saved
    }
    if = {
        limit = { NOT = { exists = scope:target_county } }
        random_held_title = {
            title_tier = county
            limit = { NOT = { this = scope:source_county } }
            save_scope_as = target_county
        }
    }
    combined_migration_effect = {
        SOURCE_COUNTY = scope:source_county
        TARGET_COUNTY = scope:target_county
        FAITH = scope:faith
        CULTURE = scope:culture
    }
}
