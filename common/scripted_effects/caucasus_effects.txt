		add_claim_on_loss = no
	}

			holder = ROOT

		}

		limit = {
			primary_title = title:k_abkhazia
		}
		title:k_georgia = { set_variable = k_georgia_abkhazia }
	}
	else_if = {
		limit = {
			primary_title = title:k_iberia
		}
		title:k_georgia = { set_variable = k_georgia_iberia }
	}
	else_if = {
		limit = {
			primary_title = title:k_kakheti
		}
		title:k_georgia = { set_variable = k_georgia_kakheti }
	}

	resolve_title_and_vassal_change = scope:title_change
	set_primary_title_to = title:k_georgia

	title:k_abkhazia = { add_to_list = georgian_kingdoms }
	title:k_iberia = { add_to_list = georgian_kingdoms }
	title:k_kakheti = { add_to_list = georgian_kingdoms }

	hidden_effect = {
		every_in_list = {
			list = georgian_kingdoms
			limit = { holder = ROOT }
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_duchy
				}
				set_de_jure_liege_title = title:k_georgia
			}

}

#Pirate Thamade Duel Effect (Meant for usage in event "caucasianroa.100")
# Requires scopes defined:
# - scope:raid_sponsor
# - scope:pirate_thamade
# - scope:raid_victim
# - scope:bone_blessing
# Usage:
# duel_pirate_thamade_raid = yes
duel_pirate_thamade_raid = {

	#Do the duel to see if detected BEFORE so the variable is set when the event is called
	duel_pirate_thamade_raid_detected = yes

	#Determine if duel should be against ruler, or their marshal
	if = {
		limit = {
			exists = scope:raid_victim.cp:councillor_marshal
			scope:raid_victim.cp:councillor_marshal.martial > scope:raid_victim.martial
		}
		scope:raid_victim.cp:councillor_marshal = { save_scope_as = duel_target }
	}
	else = {
		scope:raid_victim = { save_scope_as = duel_target }
	}

	scope:pirate_thamade = {
		duel = {
			skill = martial
			target = scope:duel_target

			# SUCCESS
			50 = {
				desc = caucasianroa.100.duel.success.d
				compare_modifier = {
					value = scope:duel_value
					multiplier = 1.25
					min = -35
				}

				# Bone Blessing
				modifier = {
					scope:bone_blessing = yes
					add = 10
				}

				# Effects:
				show_as_tooltip = { custom_tooltip = caucasianroa.100.duel.success.t }
				# Trigger Event
				scope:raid_sponsor = {
					trigger_event = {
						id = caucasianroa.101
						days = 7
					}
				}
				scope:raid_victim = {
					trigger_event = {
						id = caucasianroa.102
						days = 7
					}
				}
			}

			# FAILURE
			50 = {
				desc = caucasianroa.100.duel.failure.d
				compare_modifier = {
					value = scope:duel_value
					multiplier = -1.25
					min = -35
				}

				# Bone Blessing
				modifier = {
					scope:bone_blessing = no
					add = 10
				}

				# Effects
				show_as_tooltip = { custom_tooltip = caucasianroa.100.duel.failure.t }
				scope:raid_sponsor = {
					trigger_event = {
						id = caucasianroa.103
						days = 7
					}
				}
				scope:raid_victim = {
					trigger_event = {
						id = caucasianroa.104
						days = 7
					}
				}
			}
		}
	}
}
