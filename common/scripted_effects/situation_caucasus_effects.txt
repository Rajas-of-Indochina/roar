auto_assign_caucasus_stance = {
	# hard rules for specific titles
	if = {
		limit = {
			OR = {
				has_title = title:e_byzantium
				has_title = title:h_eastern_roman_empire
				has_title = title:h_roman_empire
			}
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			OR = {
				has_title = title:e_persia
				has_title = title:e_arabia
				has_title = title:h_dar_al_islam
			}
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	# Go by faith first
	else_if = {
		limit = {
			faith = { faith_is_pro_roman_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			faith = { faith_is_pro_perso_arabic_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	else_if = {
		limit = {
			faith = { faith_is_autonomist_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = autonomist COOLDOWN = no }
	}
	# Then by culture if faith doesn't do it
	else_if = {
		limit = {
			culture = { culture_is_pro_roman_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			culture = { culture_is_pro_perso_arabic_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	else_if = {
		limit = {
			culture = { culture_is_autonomist_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = autonomist COOLDOWN = no }
	}
	else = {
		set_caucasus_alignment = { FLAG = interloper COOLDOWN = no }
	}
}

# just a shortcut
# FLAG options: pro_roman pro_perso_arabic autonomist interloper
set_caucasus_alignment = {
	set_variable = {
		name = caucasus_alignment
		value = flag:$FLAG$
	}
	hidden_effect = { recalculate_participant_group = situation:situation_caucasus }
	if = {
		limit = { always = $COOLDOWN$ }
		set_variable = {
			name = situation_caucasus_recently_changed_alignment
			years = 5
		}
	}

	custom_tooltip = setting_$FLAG$_alignment_tt
}

# since assigning a forced alignment only requires a flag in a var, we don't need to worry about scope types!
align_culture_effect = {
	$TARGET$ = {
		set_variable = {
			name = caucasus_alignment
			value = flag:$FLAG$
		}
	}

	custom_tooltip = {
		text = culture_setting_$FLAG$_alignment_tt
		subject = $TARGET$
	}
}

form_aluank_decision_effect = {
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:k_aluank = {
		change_title_holder = {
			holder = ROOT
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	set_primary_title_to = title:k_aluank
	custom_tooltip = {
		text = aluank_de_jure_drift_tt
		title:d_arran = { add_to_list = albanian_duchies }
		title:d_artsakh = { add_to_list = albanian_duchies }
		title:d_gardman = { add_to_list = albanian_duchies }
		title:d_hereti = { add_to_list = albanian_duchies }
		title:d_shaki = { add_to_list = albanian_duchies }
		title:d_layzan = { add_to_list = albanian_duchies }
		if = {
			limit = { completely_controls = title:d_kakheti }
			title:d_kakheti = { add_to_list = albanian_duchies }
		}
		if = {
			limit = { completely_controls = title:d_shirvan }
			title:d_shirvan = { add_to_list = albanian_duchies }
		}
		every_in_list = {
			list = albanian_duchies
			set_de_jure_liege_title = title:k_aluank
		}
	}
	custom_tooltip = aluank_de_jure_extra_tt
}

form_georgia_decision_effect = {
	custom_tooltip = georgia_de_jure_tt

	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:k_georgia = {
		change_title_holder = {
			holder = ROOT
			change = scope:title_change
		}
	}

	if = {
		limit = {
			primary_title = title:k_abkhazia
		}
		title:k_georgia = { set_variable = k_georgia_abkhazia }
	}
	else_if = {
		limit = {
			primary_title = title:k_iberia
		}
		title:k_georgia = { set_variable = k_georgia_iberia }
	}
	else_if = {
		limit = {
			primary_title = title:k_kakheti
		}
		title:k_georgia = { set_variable = k_georgia_kakheti }
	}

	resolve_title_and_vassal_change = scope:title_change
	set_primary_title_to = title:k_georgia

	title:k_abkhazia = { add_to_list = georgian_kingdoms }
	title:k_iberia = { add_to_list = georgian_kingdoms }
	title:k_kakheti = { add_to_list = georgian_kingdoms }

	hidden_effect = {
		every_in_list = {
			list = georgian_kingdoms
			limit = { holder = ROOT }
			every_in_de_jure_hierarchy = {
				limit = {
					tier = tier_duchy
				}
				set_de_jure_liege_title = title:k_georgia
			}
			if = {
				limit = { exists = yes }
				scope:founder = {destroy_title = PREV}
			}
		}

	}
}
