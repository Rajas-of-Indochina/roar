auto_assign_caucasus_stance = {
	# hard rules for specific titles
	if = {
		limit = {
			is_roman_emperor_trigger = yes
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			OR = {
				has_title = title:e_persia
				has_title = title:e_arabia
				has_title = title:h_dar_al_islam
			}
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	# Go by faith first
	else_if = {
		limit = {
			faith = { rite_is_pro_roman_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			faith = { rite_is_pro_perso_arabic_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	else_if = {
		limit = {
			faith = { rite_is_autonomist_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = autonomist COOLDOWN = no }
	}
	# Then by culture if faith doesn't do it
	else_if = {
		limit = {
			culture = { culture_is_pro_roman_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_roman COOLDOWN = no }
	}
	else_if = {
		limit = {
			culture = { culture_is_pro_perso_arabic_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = pro_perso_arabic COOLDOWN = no }
	}
	else_if = {
		limit = {
			culture = { culture_is_autonomist_trigger = yes }
		}
		set_caucasus_alignment = { FLAG = autonomist COOLDOWN = no }
	}
	else = {
		set_caucasus_alignment = { FLAG = interloper COOLDOWN = no }
	}
}

		}

			add_to_list = regions_to_notify
		}
	}
	if = {
		limit = { title_overlaps_geographical_region = { TITLE = $TITLE$  REGION = custom_lesser_armenia } }
		situation:situation_caucasus ?= {
			situation_sub_region:lesser_armenia = { trigger_sub_region_catalyst = $CATALYST$ }
		}
		geographical_region:custom_lesser_armenia = {
			add_to_list = regions_to_notify
		}
	}
	if = {
		limit = { title_overlaps_geographical_region = { TITLE = $TITLE$  REGION = custom_armenian_cilicia } }
		situation:situation_caucasus ?= {
			situation_sub_region:cilicia = { trigger_sub_region_catalyst = $CATALYST$ }
		}
		geographical_region:custom_armenian_cilicia = {
			add_to_list = regions_to_notify
		}
	}
	if = {
		limit = { title_overlaps_geographical_region = { TITLE = $TITLE$  REGION = custom_syria } }
		situation:situation_caucasus ?= {
			situation_sub_region:syria = { trigger_sub_region_catalyst = $CATALYST$ }

			situation_sub_region:jazira = { trigger_sub_region_catalyst = $CATALYST$ }
		}
		geographical_region:custom_jazira = {
			add_to_list = regions_to_notify

			type = event_toast_effect_neutral
			title = situation_caucasus_catalyst_triggered_in_title
			left_icon = $TITLE$
		}
	}
}

# just a shortcut
# FLAG options: pro_roman pro_perso_arabic autonomist interloper
set_caucasus_alignment = {
	save_scope_as = char
	set_variable = {
		name = caucasus_alignment
		value = flag:$FLAG$
	}
	custom_tooltip = {
		text = setting_$FLAG$_alignment_tt
		subject = THIS
	}
	hidden_effect = { recalculate_participant_group = situation:situation_caucasus }
	if = {
		limit = { always = $COOLDOWN$ }
		set_variable = {
			name = situation_caucasus_recently_changed_alignment
			years = 5
		}
		situation:situation_caucasus ?= {
			switch = {
				trigger = scope:char.var:caucasus_alignment
				flag:pro_roman = {
					trigger_caucasus_catalyst_in_title = {
						TITLE = scope:char.primary_title
						CATALYST = character_became_pro_roman
					}
				}
				flag:pro_perso_arabic = {
					trigger_caucasus_catalyst_in_title = {
						TITLE = scope:char.primary_title
						CATALYST = character_became_pro_perso_arabic
					}
				}
				flag:autonomist = {
					trigger_caucasus_catalyst_in_title = {
						TITLE = scope:char.primary_title
						CATALYST = character_became_autonomist
					}
				}
			}
		}
	}
}

	}
	create_title_and_vassal_change = {
		type = created
		save_scope_as = title_change
		add_claim_on_loss = no
	}
	title:k_armenian_cilicia = {
		change_title_holder = {
			holder = ROOT
			change = scope:title_change
		}
	}
	resolve_title_and_vassal_change = scope:title_change
	set_primary_title_to = title:k_armenian_cilicia

			situation_sub_region:cilicia = {

			situation_sub_region:cilicia = {
				trigger_sub_region_catalyst = formed_armenian_cilicia_pro_perso_arabic

	}
	else_if = {
		limit = { top_participant_group:situation_caucasus ?= { participant_group_type = autonomist } }

			situation_sub_region:cilicia = {
				trigger_sub_region_catalyst = formed_armenian_cilicia_autonomist
			}
		}
	}
}
