create_or_add_to_stateless_confederation_effect = {
	if = {
		limit = {
			is_confederation_member = yes
		}
		confederation = { add_confederation_member_house = $TARGET$.house }
	}
	else = {
		#You two are in the same duchy, name it after that
		if = {
			limit = {
				exists = $TARGET$.capital_county
				capital_county.duchy = {
					any_de_jure_county = {
						this = $TARGET$.capital_county
					}
				}
			}
			capital_county = {
				duchy = {
					save_scope_as = capital_duchy
				}
			}
		}
		#You are a duke, name it after your duchy
		else_if = {
			limit = {
				primary_title = {
					tier = tier_duchy
					any_de_jure_county = {
						holder = $TARGET$
					}
				}
			}
			primary_title = {
				save_scope_as = capital_duchy
			}
		}
		#Name after your capital duchy
		else = {
			capital_county = {
				duchy = {
					save_scope_as = capital_duchy
				}
			}
		}
		create_confederation = {
			name = CONFEDERATION_DUCHY_NAME
			type = confederation_type:stateless_confederation
			leader = house
		}
		scope:new_confederation = {
			confederation = { add_confederation_member_house = $TARGET$.house }
		}
	}
}

# ROOT - character handing the county off
# scope:new_count - Character to be given the county
# scope:title - the county being handed off
give_commune_county_effect = {
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:title = {
		change_title_holder = {
			holder = scope:new_count
			change = scope:change
			take_baronies = yes
		}
	}
	resolve_title_and_vassal_change = scope:change

	hidden_effect = {
		scope:new_count = {
			change_government = anarchist_government
			add_realm_law_skip_effects = single_heir_dynasty_house
		}
	}

	# Opinion boost for new vassal
	scope:new_count = {
		add_opinion = {
			target = root
			modifier = stateless_given_county_opinion
		}
	}

	create_or_add_to_stateless_confederation_effect = { TARGET = scope:new_count }
	
}

convert_stateless_to_tribal_effect = {
	change_government = tribal_government
	remove_realm_law = stateless_authority_0
	add_realm_law_skip_effects = tribal_authority_0
}

convert_stateless_to_wanua_effect = {
	change_government = wanua_government
	remove_realm_law = stateless_authority_0
	add_realm_law_skip_effects = tribal_authority_0
}

centralize_stateless_confederation_effect = {
	capital_county = {
		# get the list if counties in your shiny new duchy
		duchy = {
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_county }
				add_to_list = duchy_counties
			}
		}
	}
	every_confederation_member = {
		limit = {
			# exclude yourself
			NOT = { THIS = ROOT }
		}
		add_to_list = future_tributaries
	}
	# destroy confederation, we no longer need it
	if = {
		limit = { has_cultural_parameter = heritage_group_austronesian }
		convert_stateless_to_wanua_effect = yes
	}
	else = {
		convert_stateless_to_tribal_effect = yes
	}
	# grant the duchy
	create_title_and_vassal_change = {
		type = created
		save_scope_as = duchy
		add_claim_on_loss = no
	}
	capital_county.duchy = {
		change_title_holder = {
			holder = ROOT
			change = scope:duchy
		}
	}
	resolve_title_and_vassal_change = scope:duchy
	# then hand over the counties
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = counties
		add_claim_on_loss = no
	}
	every_in_list = {
		list = duchy_counties
		change_title_holder = {
			holder = ROOT
			change = scope:counties
		}
	}
	resolve_title_and_vassal_change = scope:counties
	# convert former confederates into tributaries
	every_in_list = {
		list = future_tributaries
		save_scope_as = new_tributary
		start_tributary_interaction_effect = {
			TRIBUTARY = scope:new_tributary
			SUZERAIN = ROOT
		}
	}
}

refill_maa_with_barter_goods_effect = {
	save_scope_as = laamp_temp
	while = {
		limit = {
			scope:actor.barter_goods > 0
			any_maa_regiment = {
				maa_regiments_valid_to_refill_trigger = yes
			}
		}
		random_maa_regiment = {
			limit = {
				maa_regiments_valid_to_refill_trigger = yes
			}
			save_scope_as = maa_scope
			save_scope_value_as = {
				name = current_barter_goods_max_value_scope
				value = scope:actor.barter_goods
			}
			scope:actor = {
				remove_barter_goods = {
					add = scope:maa_scope.replenishable_troops_value
					divide = gold_refill_value
					max = scope:current_barter_goods_max_value_scope
				}
			}
			change_maa_troops_count = replenishable_troops_with_barter_goods_value
		}
	}
}
