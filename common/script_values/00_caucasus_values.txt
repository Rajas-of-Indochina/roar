#CALL FROM THE SCOPE OF THE CHARACTER TO DERIVE THE VALUE FROM
roa_pirate_target_reward_value = {
	add = {
		value = this.gold
		multiply = 0.3

		if = {
			limit = { highest_held_title_tier = tier_county }
			min = 25
			max = 175
		}
		else_if = {
			limit = { highest_held_title_tier = tier_duchy }
			min = 50
			max = 250
		}
		else_if = {
			limit = { highest_held_title_tier = tier_kingdom }
			min = 100
			max = 400
		}
		else_if = {
			limit = { highest_held_title_tier = tier_empire }
			min = 150
			max = 575
		}
		else_if = {
			limit = { highest_held_title_tier = tier_hegemony }
			min = 200
			max = 800
		}
	}

	if = {
		limit = {
			has_treasury = yes
		}
		#Make slightly less impact on personal gold because there's a second pot to grab from which is potentially very big
		multiply = 0.7 # Reduces 0.3 -> 0.21
		add = {
			value = this.treasury
			multiply = 0.20
			if = {
				limit = { highest_held_title_tier = tier_county }
				min = 25
				max = 175
			}
			else_if = {
				limit = { highest_held_title_tier = tier_duchy }
				min = 50
				max = 250
			}
			else_if = {
				limit = { highest_held_title_tier = tier_kingdom }
				min = 100
				max = 400
			}
			else_if = {
				limit = { highest_held_title_tier = tier_empire }
				min = 150
				max = 575
			}
			else_if = {
				limit = { highest_held_title_tier = tier_hegemony }
				min = 200
				max = 800
			}
		}
	}

	# Variance
	multiply = {
		value = {
			integer_range = {
				min = 90
				max = 110
			}
		}
		divide = 100
	}

	round = yes
}
