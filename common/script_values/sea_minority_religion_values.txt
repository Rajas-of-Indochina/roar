# Modify (multiply) the doctrine cost when creating a faith.
# The returned value is interpreted as a multiplicative modifier (e.g. 0.5 means +50% cost).
# scope:character = the creation character
# scope:old_faith = the faith the character is converting from
faith_creation_cost_mult = {
	subtract = {
		value = scope:old_faith.faith_conversion_fervor_mult
		desc = "faith_creation_cost_fervor_impact"
		format = "[Faith.GetName]"
	}
	if = {
		limit = {
			OR ={
				AND = {
					scope:character = {
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = roman_faith_creation_cost_cheaper
						}
					}
					armenian_struggle_would_be_roman_faith_trigger = yes
				}
			}
			OR ={
				AND = {
					scope:character = {
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = persian_faith_creation_cost_cheaper
						}
					}
					armenian_struggle_would_be_persian_faith_trigger = yes
				}
			}
			OR ={
				AND = {
					scope:character = {
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = armenian_faith_creation_cost_cheaper
						}
					}
					armenian_struggle_would_be_local_faith_trigger = yes
				}
			}
		}
		subtract = {
			value = 0.25
			desc = "faith_creation_cost_struggle_phase_impact"
			format = "BREAKDOWN_FORMAT_PIETY_PERCENTAGE_POSITIVE_IS_BAD"
		}
	}
	min = {
		value = -0.75
	}
}

num_same_faith_counties = {
	value = 0
	every_realm_county = {
		limit = { faith = PREV.faith }
		add = 1
	}
}

num_faith_counties = {
	value = 0
	every_county = {
		limit = { faith = PREV }
		add = 1
	}
}

circassian_alignment_threshold = {
	value = 0
	add = faith.num_faith_counties
	divide = 2
	add = 1
}
