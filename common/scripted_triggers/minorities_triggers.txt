county_has_small_faith_minority = {
    has_variable_list = faith_minorities_small
    variable_list_size = {
        name = faith_minorities_small
        value > 0
    }
}

county_has_large_faith_minority = {
    has_variable_list = faith_minorities_large
    variable_list_size = {
        name = faith_minorities_large
        value > 0
    }
}

county_has_faith_minority = {
    OR = {
        county_has_large_faith_minority = yes
        county_has_small_faith_minority = yes
    }
}

county_has_specific_small_faith_minority = {
    is_target_in_variable_list = {
        name = faith_minorities_small
        target = $FAITH$
    }
}
county_has_specific_large_faith_minority = {
    is_target_in_variable_list = {
        name = faith_minorities_large
        target = $FAITH$
    }
}
county_has_specific_faith_minority = {
    OR = {
        county_has_specific_small_faith_minority = { FAITH = $FAITH$ }
        county_has_specific_large_faith_minority = { FAITH = $FAITH$ }
    }
}

# TRIGGER is ANY simple scripted trigger (takes yes, no or a scope)
# CHECK is the thing being checked by the trigger (the yes, no or scope)
# you could maybe get crazier with it but don't come crying to me
# TYPE is faith or culture
# SIZE is the minority size (small or large)
# call from county scope
county_has_type_minority_with_trigger = {
	any_in_list = {
		variable = $TYPE$_minorities_$SIZE$
		$TRIGGER$ = $CHECK$
	}
}

county_has_type_minority_trigger_not = {
	any_in_list = {
		variable = $TYPE$_minorities_$SIZE$
		NOT = { $TRIGGER$ = $CHECK$ }
	}
}

county_has_expellable_culture_trigger = {
	any_in_list = {
		variable = culture_minorities_$SIZE$
		NOT = { has_same_culture_heritage = $CULTURE$ }
	}
}

county_has_expellable_faith_trigger = {
	$FAITH$ = { save_temporary_scope_as = the_faith }
	any_in_list = {
		variable = faith_minorities_$SIZE$
		save_temporary_scope_as = faith_temp
		"scope:the_faith.faith_hostility_level(scope:faith_temp)" >= faith_hostile_level
	}
}

	$CULTURE$ = {
		cultural_acceptance = {
			# they don't need to love them but they need to not actively hate them
			target = $TARGET$.culture
			value > 10
		}
	}

}

can_have_minority_laws_trigger = {
	NOT = {
		government_has_flag = government_is_nomadic
		government_has_flag = government_is_stateless
		government_has_flag = government_is_herder
		government_has_flag = cannot_be_vassal_or_liege
	}
	is_landed = yes
}
