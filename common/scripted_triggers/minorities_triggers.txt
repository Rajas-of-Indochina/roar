county_has_small_rite_minority = {
    has_variable_list = rite_minorities_small
    variable_list_size = {
        name = rite_minorities_small
        value > 0
    }
}

county_has_large_rite_minority = {
    has_variable_list = rite_minorities_large
    variable_list_size = {
        name = rite_minorities_large
        value > 0
    }
}

county_has_rite_minority = {
    OR = {
        county_has_large_rite_minority = yes
        county_has_small_rite_minority = yes
    }
}

county_has_specific_small_rite_minority = {
    is_target_in_variable_list = {
        name = rite_minorities_small
        target = $RITE$
    }
}

		variable = rite_minorities_small

	}
}

county_has_specific_large_rite_minority = {

        name = rite_minorities_large

    }
}

county_has_specific_large_faith_minority = {
	any_in_list = {
		variable = rite_minorities_large
		faith = $FAITH$
	}
}

county_has_specific_rite_minority = {

        county_has_specific_small_rite_minority = { RITE = $RITE$ }
        county_has_specific_large_rite_minority = { RITE = $RITE$ }

county_has_specific_faith_minority = {

# TRIGGER is ANY simple scripted trigger (takes yes, no or a scope)
# CHECK is the thing being checked by the trigger (the yes, no or scope)
# you could maybe get crazier with it but don't come crying to me
# TYPE is rite or culture
# SIZE is the minority size (small or large)
# call from county scope
county_has_type_minority_with_trigger = {
	any_in_list = {
		variable = $TYPE$_minorities_$SIZE$
		$TRIGGER$ = $CHECK$
	}
}

county_has_type_minority_trigger_not = {
	any_in_list = {
		variable = $TYPE$_minorities_$SIZE$
		NOT = { $TRIGGER$ = $CHECK$ }
	}
}

county_has_expellable_culture_trigger = {
	any_in_list = {
		variable = culture_minorities_$SIZE$
		NOT = { has_same_culture_heritage = $CULTURE$ }
	}
}

county_has_expellable_rite_trigger = {
	$RITE$ = { save_temporary_scope_as = the_rite }
	any_in_list = {
		variable = rite_minorities_$SIZE$
		save_temporary_scope_as = rite_temp
		"scope:the_rite.faith.faith_hostility_level(scope:rite_temp)" >= faith_hostile_level
	}
}

		}
	}
}

# For minority expulsion (chaplain)
task_expel_religious_minority_valid_county_trigger = {
	is_landless_type_title = no
	save_temporary_scope_as = county_check

	trigger_if = {
		limit = {
			scope:councillor_liege = {
				is_alive = yes
				has_variable = hold_court_8120_religion_block
			}
		}
		custom_tooltip = {
			text = hold_court_8120_conversion_tt
			faith != scope:councillor_liege.var:hold_court_8120_religion_block
		}
	}
	custom_description = { # Vassal Contract forbids meddling by liege
		text = "is_protected_via_contract_self_or_liege"
		subject = holder
		# don't "protect" same faith in edge cases
		trigger_if = {
			limit = {
				scope:county_check.holder = { is_vassal_or_below_of = scope:councillor_liege }
				scope:councillor_liege.faith != scope:county_check.holder.faith
			}
			holder = {
				NOR = {
					vassal_contract_has_flag = religiously_protected
					any_liege_or_above = {
						is_vassal_or_below_of = scope:councillor_liege
						vassal_contract_has_flag = religiously_protected
					}
				}
			}
		}
	}
	trigger_if = {
		limit = {	
			var:cannot_be_converted_by_value ?= scope:councillor_liege		
		}
		custom_tooltip = {
			text = promised_to_not_convert_county	
			NOT = {
				var:cannot_be_converted_by_value ?= scope:councillor_liege	
			}
		}
	}
	trigger_if = { # AI won't expel from counties that aren't already their own faith
		limit = {
			is_ai = yes
		}
		OR = {
			faith = scope:councillor_liege.faith
			faith = scope:councillor.faith
		}
	}
	county_has_rite_minority = yes
}
