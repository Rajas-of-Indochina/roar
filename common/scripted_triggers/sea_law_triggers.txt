### New
can_keep_nasrani_partition_law_trigger = {
	trigger_if = {
		limit = {
			has_realm_law = nasrani_partition_law
		}
		can_have_nasrani_partition_law_trigger = yes
	}
}

can_have_nasrani_partition_law_trigger = {
	custom_tooltip = {
		text = can_have_nasrani_partition_law_trigger_false_tt
		OR = {
			faith = { has_doctrine = special_doctrine_nasrani_partition_succession_law }
			culture = { has_cultural_parameter = can_enact_nasrani_partition_succession_law }
		}
	}
	can_have_confederate_partition_succession_law_trigger = yes
}


### Overwrites of Vanilla

	realm_law_use_stateless_authority = no

can_have_partition_succession_law_trigger = {
	OR = {
		culture = { has_innovation = innovation_hereditary_rule }
		historical_succession_access_partition_succession_law_trigger = yes
	}
	custom_description = {
		text = succession_laws_must_have_organised_government_tooltip
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_japan_feudal
		}
	}
}

can_have_high_partition_succession_law_trigger = {
	trigger_if = {
		limit = {
			NOT = { 
				culture = { has_cultural_parameter = can_enact_high_partition_succession_law }
			}
		}
		OR = {
			historical_succession_access_high_partition_succession_law_trigger = yes
			culture = { has_innovation = innovation_heraldry }
		}
	}
	custom_description = {
		text = succession_laws_must_have_organised_government_tooltip
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_japan_feudal
		}
	}
}
