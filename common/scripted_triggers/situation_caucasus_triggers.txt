# called from a faith scope
faith_is_pro_roman_trigger = {
	custom_tooltip = {
		text = faith_is_pro_roman_trigger_tt
		subject = THIS
		trigger_if = {
			limit = {
				OR = {
					# var:caucasus_alignment ?= flag:pro_roman
					var:caucasus_alignment ?= flag:pro_perso_arabic
					var:caucasus_alignment ?= flag:autonomist
				}
			}
			always = no
		}
		trigger_else = {
			OR = {
				# The Roman Church
				this = faith:orthodox
				religious_head_title = faith:orthodox.religious_head_title
				# The OTHER Roman Church
				this = faith:catholic
				religious_head_title = faith:catholic.religious_head_title
				# the Iconoclasts
				this = faith:iconoclast
				religious_head_title = faith:iconoclast.religious_head_title
				# Hellenic Paganism
				religion_tag = hellenism_religion
				# also include a check for faiths marked pro-Roman by flag
				var:caucasus_alignment ?= flag:pro_roman
			}
		}
	}
}

faith_is_pro_perso_arabic_trigger = {
	custom_tooltip = {
		text = faith_is_pro_perso_arabic_trigger_tt
		subject = THIS
		trigger_if = {
			limit = {
				OR = {
					var:caucasus_alignment ?= flag:pro_roman
					# var:caucasus_alignment ?= flag:pro_perso_arabic
					var:caucasus_alignment ?= flag:autonomist
				}
			}
			always = no
		}
		trigger_else = {
			OR = {
				# Zoroastrians (with exceptions)
				AND = {
					religion = religion:zoroastrianism_religion
					NOT = {
						# Certain Zoroastrian sects
						this = faith:barhailism
						this = faith:urartuism
						this = faith:mazdakism
						this = faith:khurramism
					}
				}
				# Nestorians - this will be handled by flag
				# religious_head_title = faith:nestorian.religious_head_title
				# Muslims
				religion = religion:islam_religion
				# explicitly marked pro-perso-arabic faiths
				var:caucasus_alignment ?= flag:pro_perso_arabic
			}
		}
	}
}

			always = no

				this = culture:roman

		}
	}
}
