# called from a rite scope
rite_is_pro_roman_trigger = {
	custom_tooltip = {
		text = rite_is_pro_roman_trigger_tt
		subject = THIS
		trigger_if = {
			limit = {
				OR = {
					# var:caucasus_alignment ?= flag:pro_roman
					var:caucasus_alignment ?= flag:pro_perso_arabic
					var:caucasus_alignment ?= flag:autonomist
				}
			}
			always = no
		}
		trigger_else = {
			OR = {
				# The Roman Church
				this.faith = faith:orthodox
				# The OTHER Roman Church
				this.faith = faith:catholic
				# The as-yet-undivided Church
				this.faith = faith:christian_faith
				# the Iconoclasts
				this.faith = faith:iconoclast
				# Hellenic Paganism
				this.faith = faith:hellenic_pagan
				# also include a check for faiths marked pro-Roman by flag
				var:caucasus_alignment ?= flag:pro_roman
			}
		}
	}
}

rite_is_pro_perso_arabic_trigger = {
	custom_tooltip = {
		text = rite_is_pro_perso_arabic_trigger_tt
		subject = THIS
		trigger_if = {
			limit = {
				OR = {
					var:caucasus_alignment ?= flag:pro_roman
					# var:caucasus_alignment ?= flag:pro_perso_arabic
					var:caucasus_alignment ?= flag:autonomist
				}
			}
			always = no
		}
		trigger_else = {
			OR = {
				# Zoroastrians (with exceptions)
				AND = {
					religion = religion:zoroastrianism_religion
					NOT = {
						# Certain Zoroastrian sects
						faith = faith:barhailism
						faith = faith:urartuism
						faith = faith:mazdakism
						faith = faith:khurramism
					}
				}
				# Nestorians - this will be handled by flag
				# religious_head_title = faith:nestorian.religious_head_title
				# Muslims
				religion = religion:islam_religion
				faith = faith:miaphysitism
				# explicitly marked pro-perso-arabic faiths
				var:caucasus_alignment ?= flag:pro_perso_arabic
			}
		}
	}
}

title_overlaps_geographical_region = {
	trigger_if = {
		limit = {
			OR = {
				is_nomad_title = yes
				is_noble_family_title = yes
				is_figurehead_title = yes
				is_landless_type_title = yes
			}
			$TITLE$.holder = { exists = yes }
			$TITLE$.domicile = { exists = yes }
		}
		$TITLE$.holder.domicile.domicile_location = { geographical_region = $REGION$ }
	}
	trigger_else_if = {
		limit = { $TITLE$ = { is_titular = yes } }
		$TITLE$.title_capital_county.title_province = { geographical_region = $REGION$ }
	}
	trigger_else_if = {
		limit = { $TITLE$.tier <= tier_county }
		$TITLE$.title_province = { geographical_region = $REGION$ }
	}
	trigger_else = {
		any_de_jure_county = {
			title_province = { geographical_region = $REGION$ }
		}
	}
}
