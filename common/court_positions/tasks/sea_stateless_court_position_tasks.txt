stateless_diplomacy_court_position_task_cohesion = {
	court_position_types = { stateless_diplomacy_court_position }

	court_position_asset = {
		animation = debating
	}

	cost = {
		round = no
		influence = {
			value = monthly_court_position_task_cost
			multiply = {
				value = 1
				add = {
					value = scope:liege.confederation.cohesion
					divide = 20
				}
			}
		}
	}

	on_start = {
		scope:liege = {
			set_variable = fostering_cohesion
		}
		custom_tooltip = stateless_diplomacy_court_position_task_cohesion_tt
		custom_tooltip = stateless_diplomacy_court_position_task_cohesion_monthly_tt
	}

	on_end = {
		scope:liege = {
			remove_variable = fostering_cohesion
		}
	}

	on_monthly = {
		# we can put some cohesion-changing events here later
	}

	ai_will_do = {
		value = 0
		if = {
			limit = { scope:liege = { is_confederation_member = yes } }
			add = {
				# AI desire to do this declines with confederation cohesion
				value = 100
				subtract = scope:liege.confederation.cohesion
				# but also rises (more slowly) with liege greed because we want ambitious, greedy people to try and centralize
				add = {
					value = scope:liege.ai_greed
					multiply = 0.5
					min = 0
				}
			}
			
		}
	}
}

stateless_diplomacy_court_position_task_harmony = {
	court_position_types = { stateless_diplomacy_court_position }

	court_position_asset = {
		animation = personality_compassionate
	}

	cost = {
		round = no
		influence = {
			value = monthly_court_position_task_cost
		}
		prestige = {
			value = monthly_court_position_task_cost
		}
	}

	scaling_employer_modifiers = {
		terrible = {
			learn_language_scheme_phase_duration_add = -4
		}
		poor = {
			learn_language_scheme_phase_duration_add = -8
		}
		average = {
			learn_language_scheme_phase_duration_add = -12
		}
		good = {
			learn_language_scheme_phase_duration_add = -16
		}
		excellent = {
			learn_language_scheme_phase_duration_add = -20
		}
	}

	on_start = {
		custom_tooltip = stateless_diplomacy_court_position_task_harmony_tt
	}

	on_monthly = {
		# if you aren't the same culture as your county you get chances to convert to it for free
		# or raise the acceptance between your cultures
		# or hybridize the cultures if acceptance is high enough
		# or more rarely a chance to convert the county to your culture

		# if you ARE the same culture as your county you get to raise acceptance with an adjacent county culture
		# and possibly hybridize with it
	}

	is_valid_showing_failures_only = {
		OR = {
			custom_tooltip = {
				text = different_culture_from_county_tt
				scope:liege = {
					any_realm_county = { NOT = { culture = scope:liege.culture } }
				}
			}
			custom_tooltip = {
				text = different_neighboring_culture_tt
				scope:liege = {
					any_realm_county = {
						any_neighboring_county = {
							NOT = {
								culture = scope:liege.culture
							}
						}
					}
				}
			}
		}
	}

	ai_will_do = {
		value = 50
		subtract = scope:liege.ai_zeal
		add = scope:liege.ai_compassion
	}
	
}
