situation_caucasus_imperial_support_interaction = {
	category = interaction_category_diplomacy
	icon = religious_claim
	interface_priority = 35

	desc = situation_caucasus_imperial_support_interaction_desc
	target_type = title
	target_filter = secondary_recipient_domain_titles
	redirect = {
		if = {
			limit = {
				exists = title:e_byzantium.holder
			}
			scope:recipient = {
				save_scope_as = secondary_recipient
			}
			title:e_byzantium.holder = {
				save_scope_as = recipient
			}
		}
		else_if = {
			limit = {
				exists = title:h_eastern_roman_empire.holder
			}
			scope:recipient = {
				save_scope_as = secondary_recipient
			}
			title:h_eastern_roman_empire.holder = {
				save_scope_as = recipient
			}
		}
		else_if = {
			limit = {
				exists = title:h_roman_empire.holder
			}
			scope:recipient = {
				save_scope_as = secondary_recipient
			}
			title:h_roman_empire.holder = {
				save_scope_as = recipient
			}
		}
	}

	has_valid_target_showing_failures_only = {
		# Must be a county or duchy that falls at least partly inside a sub-region where this interaction is allowed
		scope:target = {
			tier >= tier_county
			tier <= tier_duchy
			is_noble_family_title = no
			any_in_de_jure_hierarchy = {
				continue = { tier > tier_county }
				any_county_situation_sub_region = {
					any_situation_sub_region_participant_group = {
						participant_group_has_character = scope:actor
						has_participant_group_parameter = unlocks_imperial_support_interaction
					}
				}
			}
		}
	}

	is_shown = {
		# Emperor must exist
		OR = {
			exists = title:e_byzantium.holder
			exists = title:h_eastern_roman_empire.holder
			exists = title:h_roman_empire.holder
		}
		# Requester must be Pro-Roman
		scope:actor = { top_participant_group:situation_caucasus ?= { participant_group_type = pro_roman } }
		# Target must NOT be Pro-Roman
		scope:secondary_recipient = {
			NOT = { top_participant_group:situation_caucasus ?= { participant_group_type = pro_roman } }
			is_landed_or_landless_administrative = yes
			primary_title = { tier >= tier_county }
		}
		# this is just here to catch super bizarre edge cases
		NOR = {
			scope:actor = scope:secondary_recipient
			scope:secondary_recipient = scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = { # Can't request on titles you already have a pressed claim on
			limit = {
				exists = scope:target
				scope:actor = {
					has_strong_claim_on = scope:target
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_has_pressed_claim_already"
				always = no
			}
		}
		trigger_if = { # Can't request claims on baronies
			limit = {
				OR = {
					scope:recipient.highest_held_title_tier = tier_barony
					AND = {
						exists = scope:target
						scope:target = {
							tier = tier_barony
						}
					}
				}
			}
			custom_description = {
				text = "hof_ask_for_claim_no_baronies"
				always = no
			}
		}
		scope:actor = { # Variable pretige/influence cost depending on tier
			trigger_if = {
				limit = {
					exists = scope:target
				}
				OR = {
					AND = {
						government_has_flag = government_has_influence
						influence >= claim_piety_value
					}
					prestige >= claim_piety_value
				}
				
			}
			NOT = {
				is_at_war_with = scope:recipient
			}
		}
		scope:secondary_recipient = { NOT = { has_strong_hook = scope:actor } }
	}

	cost = {
		influence = {
			value = 0
			if = {
				limit = { scope:actor = { government_has_flag = government_has_influence } }
				add = claim_piety_value
			}
		}
		prestige = {
			value = 0
			if = {
				limit = { NOT = { scope:actor = { government_has_flag = government_has_influence } } }
				add = claim_piety_value
			}
		}
	}

	ai_targets = {
		ai_recipients = neighboring_rulers
		ai_recipients = peer_vassals
		ai_recipients = liege
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 108
		duchy = 36
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	ai_potential = {
		is_playable_character = yes
		OR = {
			ai_greed >= 50
			ai_has_warlike_personality = yes
		}
		OR = {
			AND = {
				government_has_flag = government_has_influence
				influence >= major_piety_value
			}
			prestige >= major_piety_value
		}
		OR = {
			exists = title:e_byzantium.holder
			exists = title:h_eastern_roman_empire.holder
			exists = title:h_roman_empire.holder
		}
	}

	ai_will_do = {
		base = 0

		modifier = {
			add = scope:actor.ai_greed
		}

		modifier = {
			add = scope:target.tier
		}

		modifier = {
			add = 500
			has_variable = conqueror
			NOR = {
				has_relation_friend = scope:secondary_recipient
				has_relation_lover = scope:secondary_recipient
			}
		}

		modifier = {
			add = -100
			scope:actor = {
				OR = {
					has_relation_friend = scope:secondary_recipient
					has_relation_lover = scope:secondary_recipient
					opinion = {
						target = scope:secondary_recipient
						value >= high_positive_opinion
					}
				}
			}
		}

		modifier = {
			factor = 0
			NOT = {
				scope:target = {
					is_neighbor_to_realm = scope:actor
				}
			}
		}
	}

	ai_accept = {
		base = 0 # Should be 0 for all interactions

		modifier = {
			scope:claim_hook ?= yes
			add = 100
			desc = SCHEME_WEAK_HOOK_USED
		}

		modifier = {
			add = -100
			desc = BASE_RELUCTANCE
		}

		opinion_modifier = { # More likely if Recipient likes Actor
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}

		modifier = { # Same language
			add = 5
			desc = speaks_same_language_interaction_reason
			trigger = {
				scope:actor = {
					knows_language_of_culture = scope:recipient.culture
				}
			}
		}
		modifier = { # Unlikely if recipient likes Secondary Recipient
			desc = AI_OPINION_RECIPIENT_REASON
			add = {
				if = {
					limit = {
						scope:recipient != scope:secondary_recipient
					}
					subtract = "scope:recipient.opinion(scope:secondary_recipient)"
				}
			}
		}

		modifier = { # Penalty for asking claims on your own vassals
			add = -50
			desc = THEY_ARE_YOUR_VASSAL
			scope:actor = {
				any_vassal_or_below = {
					this = scope:secondary_recipient
				}
			}
		}

		modifier = { # Bonus for adults to ask on children
			add = 25
			desc = THEY_ARE_A_CHILD
			scope:secondary_recipient = {
				is_adult = no
			}
			scope:actor = {
				is_adult = yes
			}
		}

		modifier = { # Penalty for children to ask on adults
			add = -25
			desc = I_AM_A_CHILD
			scope:secondary_recipient = {
				is_adult = yes
			}
			scope:actor = {
				is_adult = no
			}
		}

		modifier = { # They are of the same dynasty as the Emperor
			add = -50
			desc = THEY_ARE_MY_DYNASTY_MEMBER
			scope:recipient = {
				is_lowborn = no
				dynasty = scope:secondary_recipient.dynasty
				NOR = {
					house = scope:secondary_recipient.house
					is_close_or_extended_family_of = scope:actor
					is_cousin_of = scope:actor
					is_child_of = scope:actor
					scope:recipient.dynasty = scope:actor.dynasty
				}
			}
		}

		modifier = { # You are of the same dynasty as the Emperor
			add = 10
			desc = YOU_ARE_MY_DYNASTY_MEMBER
			scope:recipient = {
				is_lowborn = no
				dynasty = scope:actor.dynasty
				NOR = {
					is_close_or_extended_family_of = scope:actor
					is_cousin_of = scope:actor
					is_child_of = scope:actor
					scope:recipient.dynasty = scope:secondary_recipient.dynasty
				}
			}
		}

		modifier = { # They are of the same house as the Emperor
			add = -50
			desc = THEY_ARE_MY_HOUSE_MEMBER
			scope:recipient = {
				is_lowborn = no
				house = scope:secondary_recipient.house
				NOR = {
					house = scope:actor.house
					is_close_or_extended_family_of = scope:actor
					is_cousin_of = scope:actor
					is_child_of = scope:actor
					scope:recipient.house = scope:actor.house
				}
			}
		}

		modifier = { # You are of the same house as the Emperor
			add = 10
			desc = YOU_ARE_MY_HOUSE_MEMBER
			scope:recipient = {
				is_lowborn = no
				house = scope:actor.house
				NOR = {
					is_close_or_extended_family_of = scope:actor
					is_cousin_of = scope:actor
					is_child_of = scope:actor
					scope:recipient.house = scope:secondary_recipient.house
				}
			}
		}

		modifier = { # Related to the Emperor
			add = 25
			desc = YOU_ARE_MY_RELATIVE
			scope:recipient = {
				OR = {
					is_close_or_extended_family_of = scope:actor
					is_cousin_of = scope:actor
				}
				NOT = { is_child_of = scope:actor }
			}
		}

		modifier = { # Parent of the Emperor
			add = 50
			desc = YOU_ARE_MY_PARENT
			scope:recipient = {
				is_child_of = scope:actor
			}
		}

		modifier = { # Friend of the Emperor
			add = 15
			desc = WE_ARE_FRIENDS
			scope:recipient = {
				has_relation_friend = scope:actor
			}
		}

		modifier = { # Best Friend of the Emperor
			add = 20
			desc = WE_ARE_BEST_FRIENDS
			scope:recipient = {
				has_relation_best_friend = scope:actor
			}
		}

		modifier = { # Lover of the Emperor
			add = 10
			desc = WE_ARE_LOVERS
			scope:recipient = {
				has_relation_lover = scope:actor
			}
		}

		modifier = { # Rival of the Emperor
			add = -100
			desc = WE_ARE_RIVALS
			scope:recipient = {
				has_relation_rival = scope:actor
			}
		}

		modifier = { # Nemesis of the Emperor
			add = -500
			desc = WE_ARE_NEMESES
			scope:recipient = {
				has_relation_nemesis = scope:actor
			}
		}
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:actor
			object = scope:recipient
			scope:claim_hook = yes
			scope:actor = { has_strong_hook = scope:recipient }
		}
	}
	send_option = {
		is_shown = {
			scope:actor != scope:recipient
		}
		is_valid = {
			scope:actor = {
				has_usable_hook = scope:recipient
			}
		}
		flag = claim_hook
		localization = GENERIC_SPEND_A_HOOK
	}
	should_use_extra_icon = {
		scope:actor = { has_usable_hook = scope:recipient }
	}
	extra_icon = "gfx/interface/icons/character_interactions/hook_icon.dds"

	send_options_exclusive = no

	ai_min_reply_days = 1
	ai_max_reply_days = 5

	on_accept = {
		hof_ask_for_claim_interaction_effect = yes
		scope:secondary_recipient = {
			trigger_event = religious_interaction.5
		}
		scope:actor = {
			trigger_event = religious_interaction.4
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:actor
			TARGET = scope:secondary_recipient
			VALUE = minor_unity_loss
			DESC = clan_unity_granted_claim.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}
}

		trigger_if = {
			limit = { scope:actor = { top_participant_group:situation_caucasus ?= { participant_group_type = pro_roman } } }
			NOT = { scope:recipient = { top_participant_group:situation_caucasus ?= { participant_group_type = pro_roman } } }
		}
		trigger_if = {
			limit = { scope:actor = { top_participant_group:situation_caucasus ?= { participant_group_type = pro_perso_arabic } } }

		}
		trigger_if = {
			limit = { scope:actor = { top_participant_group:situation_caucasus ?= { participant_group_type = autonomist } } }
			NOT = { scope:recipient = { top_participant_group:situation_caucasus ?= { participant_group_type = autonomist } } }
		}
		# not allowed for interlopers
		NOT = { scope:actor = { top_participant_group:situation_caucasus ?= { participant_group_type = interloper } } }
		# Also not allowed for any of the faction-locked participants:
		scope:recipient = {
			NOR = {
				has_title = title:e_arabia
				has_title = title:e_persia

				has_title = title:e_byzantium
				has_title = title:h_roman_empire
				has_title = title:h_eastern_roman_empire
			}
		}
	}

	is_valid_showing_failures_only = {
		# We want to avoid the Emperor, Caliph or Shah's allies getting swayed away
		scope:recipient = {
			# general triggers
			is_at_war = no
			# Cooldown
			custom_tooltip = {
				text = fp3_change_allegiance.tt.recently_changed_mind
